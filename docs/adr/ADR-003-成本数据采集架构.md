## 一、背景

### 1.1 问题陈述

随着 Orion 平台接入的服务和项目增多，云资源成本呈现快速增长趋势。当前存在以下问题：

1. **成本黑盒**：无法按团队/项目/产品线细粒度分摊成本
2. **浪费严重**：大量闲置资源未及时发现和清理
3. **预算失控**：缺少预算预警和配额管理机制
4. **决策困难**：缺乏成本优化建议和 ROI 分析

### 1.2 决策目标

设计一套成本数据采集与分析系统，实现：
- 细粒度成本归集（团队/项目/环境维度）
- 实时成本监控与预算预警
- AI 驱动的成本优化建议
- 资源 ROI 评估与配额管理

### 2.2 数据采集流程

```
1. 资源发现
   └── 定时扫描 Kubernetes 集群
   └── 发现所有 Pod/Service/Deployment
   └── 关联团队/项目标签

2. 数据采集
   └── 每 5 分钟采集资源使用量 (CPU/Memory/Storage)
   └── 每小时同步云厂商账单数据
   └── 每日更新资源单价

3. 成本计算
   └── 资源用量 × 单价 = 资源成本
   └── 按标签分摊到团队/项目
   └── 汇总生成日报/周报/月报

4. 数据存储
   └── 明细数据存入 ClickHouse (保留 1 年)
   └── 汇总数据存入 MySQL (永久保留)
   └── 缓存热点数据到 Redis
```

# services/cost/collector.py

import asyncio
from datetime import datetime, timedelta
from typing import List, Dict
from kubernetes import client, config
import aioboto3

class CostCollector:
    """成本数据采集服务"""
    
    def __init__(self, config: dict):
        self.config = config
        self.k8s_client = None
        self.aws_client = None
        
    async def initialize(self):
        """初始化客户端"""
        config.load_incluster_config()
        self.k8s_client = client.CoreV1Api()
        
        self.aws_session = aioboto3.Session(
            aws_access_key_id=self.config['aws_access_key'],
            aws_secret_access_key=self.config['aws_secret_key']
        )
    
    async def collect_k8s_resources(self) -> List[ResourceUsage]:
        """采集 Kubernetes 资源用量"""
        usages = []
        
        # 获取所有 Pod
        pods = await self.k8s_client.list_pod_for_all_namespaces(
            watch=False,
            label_selector='team_id,project_id'  # 必须有团队/项目标签
        )
        
        for pod in pods.items:
            # 提取标签
            team_id = pod.metadata.labels.get('team_id')
            project_id = pod.metadata.labels.get('project_id')
            environment = pod.metadata.labels.get('environment', 'dev')
            
            # 获取资源用量
            metrics = await self._get_pod_metrics(pod)
            
            for container in pod.spec.containers:
                # 计算成本
                cpu_cost = self._calculate_cost(
                    'cpu', 
                    metrics.cpu_usage, 
                    pod.metadata.creation_timestamp
                )
                memory_cost = self._calculate_cost(
                    'memory',
                    metrics.memory_usage,
                    pod.metadata.creation_timestamp
                )
                
                usages.append(ResourceUsage(
                    timestamp=datetime.now(),
                    team_id=team_id,
                    project_id=project_id,
                    environment=environment,
                    resource_type='compute',
                    cpu_cores=metrics.cpu_usage,
                    memory_gb=metrics.memory_usage,
                    cpu_cost=cpu_cost,
                    memory_cost=memory_cost
                ))
        
        return usages
    
    async def collect_aws_costs(self, start_date: datetime, end_date: datetime) -> List[AWSCost]:
        """从 AWS Cost Explorer 采集成本数据"""
        costs = []
        
        async with self.aws_session.client('ce') as ce_client:
            response = await ce_client.get_cost_and_usage(
                TimePeriod={'Start': start_date.isoformat(), 'End': end_date.isoformat()},
                Granularity='DAILY',
                Metrics=['UnblendedCost'],
                GroupBy=[
                    {'Type': 'DIMENSION', 'Key': 'SERVICE'},
                    {'Type': 'TAG', 'Key': 'team_id'},
                    {'Type': 'TAG', 'Key': 'project_id'}
                ]
            )
            
            for result in response['ResultsByTime']:
                for group in result['Groups']:
                    costs.append(AWSCost(
                        date=result['TimePeriod']['Start'],
                        service=group['Keys'][0],
                        team_id=group['Keys'][1] if len(group['Keys']) > 1 else None,
                        project_id=group['Keys'][2] if len(group['Keys']) > 2 else None,
                        cost=float(group['Metrics']['UnblendedCost']['Amount'])
                    ))
        
        return costs
    
    def _calculate_cost(self, resource_type: str, usage: float, duration: timedelta) -> float:
        """计算资源成本"""
        # 从缓存获取单价
        rate = self._get_resource_rate(resource_type)
        
        # 成本 = 用量 × 单价 × 时长 (小时)
        hours = duration.total_seconds() / 3600
        return usage * rate * hours

from typing import List, Dict
import pandas as pd
from sklearn.ensemble import IsolationForest

class CostAnalyzer:
    """成本分析服务"""
    
    def analyze_trend(self, team_id: str, days: int = 30) -> TrendAnalysis:
        """分析成本趋势"""
        # 获取历史数据
        usage_data = self._get_usage_data(team_id, days)
        
        # 计算趋势
        df = pd.DataFrame(usage_data)
        df['date'] = pd.to_datetime(df['usage_date'])
        
        # 7 日移动平均
        df['cost_ma7'] = df['total_cost'].rolling(window=7).mean()
        
        # 增长率
        df['cost_growth'] = df['total_cost'].pct_change()
        
        # 预测下周成本 (简单线性回归)
        forecast = self._forecast_cost(df, days=7)
        
        return TrendAnalysis(
            team_id=team_id,
            current_daily_cost=df['total_cost'].iloc[-1],
            avg_daily_cost=df['total_cost'].mean(),
            growth_rate=df['cost_growth'].mean(),
            forecast=forecast,
            trend='up' if df['cost_growth'].iloc[-1] > 0 else 'down'
        )
    
    def detect_anomalies(self, team_id: str) -> List[Anomaly]:
        """检测成本异常"""
        # 获取历史数据
        usage_data = self._get_usage_data(team_id, 90)
        
        # 使用 Isolation Forest 检测异常
        df = pd.DataFrame(usage_data)
        
        model = IsolationForest(contamination=0.05)
        df['anomaly_score'] = model.fit_predict(df[['total_cost']])
        
        anomalies = []
        for _, row in df[df['anomaly_score'] == -1].iterrows():
            anomalies.append(Anomaly(
                date=row['usage_date'],
                actual_cost=row['total_cost'],
                expected_cost=row['total_cost'] * 0.7,  # 简化处理
                deviation_pct=(row['total_cost'] - row['total_cost'] * 0.7) / (row['total_cost'] * 0.7)
            ))
        
        return anomalies
    
    def generate_optimization_suggestions(self, team_id: str) -> List[OptimizationSuggestion]:
        """生成优化建议"""
        suggestions = []
        
        # 1. 识别闲置资源

        for resource in idle_resources:
            suggestions.append(OptimizationSuggestion(
                type='idle_resource',
                title=f"清理闲置资源：{resource.name}",
                description=f"该资源已 {resource.idle_days} 天无流量",
                expected_saving=resource.monthly_cost,
                confidence=0.9
            ))

                expected_saving=instance.potential_saving,
                confidence=0.8
            ))

from typing import List
from datetime import datetime

class BudgetAlertService:
    """预算预警服务"""
    
    async def check_budget_status(self, team_id: str) -> BudgetStatus:
        """检查团队预算状态"""
        # 获取本月预算
        budget = self._get_current_month_budget(team_id)
        
        # 计算本月已用成本
        used_cost = self._get_month_to_date_cost(team_id)
        
        # 计算使用率
        usage_pct = (used_cost / budget.budget_amount) * 100
        
        # 预测月底成本
        forecast = self._forecast_month_end_cost(team_id)
        
        return BudgetStatus(
            team_id=team_id,
            budget_amount=budget.budget_amount,
            used_cost=used_cost,
            usage_pct=usage_pct,
            forecast_cost=forecast,
            remaining_days=self._get_remaining_days_in_month(),
            status=self._determine_status(usage_pct, budget)
        )
    
    async def send_alerts(self, budget_status: BudgetStatus):
        """发送预算预警通知"""
        if budget_status.usage_pct >= budget_status.critical_threshold:
            await self._send_critical_alert(budget_status)
        elif budget_status.usage_pct >= budget_status.warning_threshold:
            await self._send_warning_alert(budget_status)
    
    async def _send_critical_alert(self, status: BudgetStatus):
        """发送严重预警 (预算超支)"""
        await self.notification_service.send(
            recipients=[status.team_owner],
            subject=f"【严重】{status.team_name} 本月预算已超支",
            content=f"""
本月预算：${status.budget_amount}
已用成本：${status.used_cost} ({status.usage_pct:.1f}%)
预测月底：${status.forecast_cost}

请立即检查资源使用情况并采取优化措施。
""",
            channels=['email', 'im']
        )
    
    async def _send_warning_alert(self, status: BudgetStatus):
        """发送警告预警 (接近预算)"""
        await self.notification_service.send(
            recipients=[status.team_owner],
            subject=f"【警告】{status.team_name} 本月预算使用已达{status.usage_pct:.1f}%",
            content=f"""
本月预算：${status.budget_amount}
已用成本：${status.used_cost} ({status.usage_pct:.1f}%)
剩余天数：{status.remaining_days} 天

建议关注成本趋势，避免超支。
""",
            channels=['im']

## 六、监控告警

### 6.1 成本监控指标

| 指标 | 采集方式 | 告警阈值 | 说明 |
|------|---------|---------|------|
| 日成本 | 每日汇总 | 环比增长 > 50% | 成本突增 |
| 预算使用率 | 实时计算 | > 80% 警告，> 100% 严重 | 预算超支风险 |
| 闲置资源数量 | 每日扫描 | > 10 | 资源浪费 |
| 优化建议采纳率 | 统计计算 | < 30% | 优化效果不佳 |

### 6.2 Prometheus 告警规则

```yaml
groups:
- name: cost_alerts
  rules:
  # 成本突增告警
  - alert: CostSpike
    expr: |
      (
        sum(delta(cost_total_daily[1d])) 
        / 
        sum(cost_total_daily offset 1d)
      ) > 0.5
    for: 1h
    labels:
      severity: warning
    annotations:
      summary: "成本较昨日增长超过 50%"
      
  # 预算超支预警
  - alert: BudgetOverrun
    expr: |
      cost_budget_usage_pct > 80
    for: 6h
    labels:
      severity: warning
    annotations:
      summary: "预算使用率超过 80%"
      
  # 预算严重超支
  - alert: BudgetCritical
    expr: |
      cost_budget_usage_pct > 100
    for: 1h
    labels:
      severity: critical
    annotations:
      summary: "预算已超支"
```
