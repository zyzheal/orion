## 二、多标签页状态同步机制

### 2.1 TabSyncManager 实现

```typescript
// utils/TabSyncManager.ts

type SyncChannel = 'auth' | 'approval' | 'notification' | 'pipeline'

interface SyncMessage<T = unknown> {
  type: string
  payload: T
  timestamp: number
  sourceTabId: string
}

interface AuthPayload {
  event: 'login' | 'logout' | 'refresh'
  token?: string
  expiresAt?: number
}

interface ApprovalPayload {
  approvalId: string
  action: 'approved' | 'rejected' | 'transferred'
  actorId: string
  newStatus: string
}

interface NotificationPayload {
  notificationId: string
  action: 'read' | 'unread' | 'delete'
}

class TabSyncManager {
  private channel: BroadcastChannel
  private tabId: string
  private listeners: Map<string, Set<(data: unknown) => void>>
  private storagePrefix = 'orion_sync_'

  constructor() {
    this.tabId = this.generateTabId()
    this.channel = new BroadcastChannel('orion_design_sync')
    this.listeners = new Map()
    this.init()
  }

  private generateTabId(): string {
    return `tab_${Date.now()}_${Math.random().toString(36).slice(2, 9)}`
  }

  private init(): void {
    // BroadcastChannel 消息监听
    this.channel.onmessage = (event: MessageEvent<SyncMessage>) => {
      if (event.data.sourceTabId === this.tabId) return
      this.handleMessage(event.data)
    }

    // localStorage 降级方案（兼容旧浏览器）
    window.addEventListener('storage', this.handleStorageChange)

    // 页面卸载时清理
    window.addEventListener('beforeunload', this.cleanup)
  }

  private handleMessage(message: SyncMessage): void {
    const handlers = this.listeners.get(message.type)
    if (!handlers) return

    handlers.forEach((handler) => {
      try {
        handler(message.payload)
      } catch (error) {
        console.error('[TabSync] Handler error:', error)
      }
    })
  }

  private handleStorageChange = (event: StorageEvent): void => {
    if (!event.key?.startsWith(this.storagePrefix)) return

    try {
      const message: SyncMessage = JSON.parse(event.newValue || '{}')
      if (message.sourceTabId === this.tabId) return
      this.handleMessage(message)
    } catch {
      // 解析失败忽略
    }
  }

  private broadcast<T>(channel: SyncChannel, type: string, payload: T): void {
    const message: SyncMessage<T> = {
      type,
      payload,
      timestamp: Date.now(),
      sourceTabId: this.tabId
    }

    // 优先使用 BroadcastChannel
    this.channel.postMessage(message)

    // 同时写入 localStorage 作为降级方案
    const storageKey = `${this.storagePrefix}${channel}_${Date.now()}`
    localStorage.setItem(storageKey, JSON.stringify(message))

    // 5 秒后清理，避免存储膨胀
    setTimeout(() => localStorage.removeItem(storageKey), 5000)
  }

  // ========== 公开 API ==========

  /** 订阅同步事件 */
  subscribe<T>(channel: SyncChannel, handler: (data: T) => void): () => void {
    const handlers = this.listeners.get(channel) || new Set()
    handlers.add(handler as (data: unknown) => void)
    this.listeners.set(channel, handlers)

    // 返回取消订阅函数
    return () => {
      const current = this.listeners.get(channel)
      current?.delete(handler as (data: unknown) => void)
    }
  }

  /** 同步认证状态 */
  syncAuth(payload: AuthPayload): void {
    this.broadcast('auth', 'auth:change', payload)

    // 同时更新当前 tab 的 token
    if (payload.event === 'logout') {
      localStorage.removeItem('orion_token')
      localStorage.removeItem('orion_token_expires')
    } else if (payload.token) {
      localStorage.setItem('orion_token', payload.token)
      if (payload.expiresAt) {
        localStorage.setItem('orion_token_expires', String(payload.expiresAt))
      }
    }
  }

  /** 同步审批状态 */
  syncApproval(payload: ApprovalPayload): void {
    this.broadcast('approval', 'approval:update', payload)
  }

  /** 同步通知已读状态 */
  syncNotification(payload: NotificationPayload): void {
    this.broadcast('notification', 'notification:update', payload)

    // 本地缓存已读状态
    if (payload.action === 'read') {
      const readKey = 'orion_read_notifications'
      const readIds = JSON.parse(localStorage.getItem(readKey) || '[]')
      if (!readIds.includes(payload.notificationId)) {
        readIds.push(payload.notificationId)
        localStorage.setItem(readKey, JSON.stringify(readIds))
      }
    }
  }

  /** 同步 Pipeline 状态 */
  syncPipeline(payload: { pipelineId: string; status: string }): void {
    this.broadcast('pipeline', 'pipeline:update', payload)
  }

  /** 获取当前 Tab ID */
  getTabId(): string {
    return this.tabId
  }

  /** 清理资源 */
  private cleanup = (): void => {
    this.channel.close()
    window.removeEventListener('storage', this.handleStorageChange)
    window.removeEventListener('beforeunload', this.cleanup)
  }

  /** 手动销毁 */
  destroy(): void {
    this.cleanup()
    this.listeners.clear()
  }
}

// 导出单例
export const tabSyncManager = new TabSyncManager()
```

```typescript
// hooks/useTabSync.ts

import { useEffect, useCallback } from 'react'
import { tabSyncManager } from '@/utils/TabSyncManager'
import { useAuthStore } from '@/stores/authStore'
import { useNotificationStore } from '@/stores/notificationStore'

export function useTabSync(): void {
  const { setToken, clearToken, refreshToken } = useAuthStore()
  const { markAsRead } = useNotificationStore()

  useEffect(() => {
    // 订阅认证状态变更
    const unsubscribeAuth = tabSyncManager.subscribe<{
      event: 'login' | 'logout' | 'refresh'
      token?: string
      expiresAt?: number
    }>('auth', (payload) => {
      if (payload.event === 'logout') {
        clearToken()
        window.location.href = '/login'
      } else if (payload.event === 'login' && payload.token) {
        setToken(payload.token, payload.expiresAt)
      } else if (payload.event === 'refresh' && payload.token) {
        refreshToken(payload.token, payload.expiresAt)
      }
    })

    // 订阅通知状态变更
    const unsubscribeNotification = tabSyncManager.subscribe<{
      notificationId: string
      action: 'read' | 'unread' | 'delete'
    }>('notification', (payload) => {
      if (payload.action === 'read') {
        markAsRead(payload.notificationId)
      }
    })

    return () => {
      unsubscribeAuth()
      unsubscribeNotification()
    }
  }, [setToken, clearToken, refreshToken, markAsRead])
}
```

```typescript
// utils/formValidation.ts

export type ValidationTrigger = 'onChange' | 'onBlur' | 'onSubmit'
export type ValidationType = 'required' | 'pattern' | 'min' | 'max' | 'custom' | 'email' | 'phone' | 'url'

export interface ValidationRule {
  type: ValidationType
  trigger?: ValidationTrigger
  message: string
  validator?: (value: unknown) => boolean | Promise<boolean>
  pattern?: RegExp
  min?: number
  max?: number
  minLength?: number
  maxLength?: number
}

export interface FieldConfig {
  label: string
  type: 'text' | 'number' | 'email' | 'password' | 'select' | 'radio' | 'checkbox' | 'textarea' | 'date' | 'file'
  placeholder?: string
  required?: boolean
  defaultValue?: unknown
  options?: { label: string; value: unknown }[]
  rules?: ValidationRule[]
  transform?: (value: unknown) => unknown
}

export interface FormConfig {
  fields: Record<string, FieldConfig>
  layout?: 'horizontal' | 'vertical' | 'inline'
  labelWidth?: number
  showRequiredMark?: boolean
}

// 内置验证器
const builtinValidators = {
  required: (value: unknown): boolean => {
    if (typeof value === 'string') return value.trim() !== ''
    if (Array.isArray(value)) return value.length > 0
    if (typeof value === 'number') return !Number.isNaN(value)
    return value !== null && value !== undefined
  },

  pattern: (value: unknown, pattern?: RegExp): boolean => {
    if (!pattern || typeof value !== 'string') return true
    return pattern.test(value)
  },

  min: (value: unknown, min?: number): boolean => {
    if (typeof value !== 'number') return true
    if (min === undefined) return true
    return value >= min
  },

  max: (value: unknown, max?: number): boolean => {
    if (typeof value !== 'number') return true
    if (max === undefined) return true
    return value <= max
  },

  minLength: (value: unknown, min?: number): boolean => {
    if (typeof value !== 'string') return true
    if (min === undefined) return true
    return value.length >= min
  },

  maxLength: (value: unknown, max?: number): boolean => {
    if (typeof value !== 'string') return true
    if (max === undefined) return true
    return value.length <= max
  },

  email: (value: unknown): boolean => {
    if (typeof value !== 'string') return false
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
    return emailRegex.test(value)
  },

  phone: (value: unknown): boolean => {
    if (typeof value !== 'string') return false
    const phoneRegex = /^1[3-9]\d{9}$/
    return phoneRegex.test(value)
  },

  url: (value: unknown): boolean => {
    if (typeof value !== 'string') return false
    try {
      new URL(value)
      return true
    } catch {
      return false
    }
  }
}

// 验证引擎
export class FormValidator {
  private config: FormConfig

  constructor(config: FormConfig) {
    this.config = config
  }

  async validateField(fieldName: string, value: unknown): Promise<string | null> {
    const field = this.config.fields[fieldName]
    if (!field) return null

    const rules = field.rules || []

    // 如果字段标记为 required 但没有 rules，添加默认 required 规则
    if (field.required && !rules.some(r => r.type === 'required')) {
      rules.unshift({ type: 'required', message: `${field.label}不能为空` })
    }

    for (const rule of rules) {
      const error = await this.validateRule(value, rule)
      if (error) return error
    }

    return null
  }

  private async validateRule(value: unknown, rule: ValidationRule): Promise<string | null> {
    const { type, message, validator, pattern, min, max, minLength, maxLength } = rule

    // 使用自定义验证器
    if (validator) {
      const isValid = await validator(value)
      return isValid ? null : message
    }

    // 使用内置验证器
    let isValid = true

    switch (type) {
      case 'required':
        isValid = builtinValidators.required(value)
        break
      case 'pattern':
        isValid = builtinValidators.pattern(value, pattern)
        break
      case 'min':
        isValid = builtinValidators.min(value, min)
        break
      case 'max':
        isValid = builtinValidators.max(value, max)
        break
      case 'minLength':
        isValid = builtinValidators.minLength(value, minLength)
        break
      case 'maxLength':
        isValid = builtinValidators.maxLength(value, maxLength)
        break
      case 'email':
        isValid = builtinValidators.email(value)
        break
      case 'phone':
        isValid = builtinValidators.phone(value)
        break
      case 'url':
        isValid = builtinValidators.url(value)
        break
      case 'custom':
        isValid = true // custom 类型必须提供 validator
        break
    }

    return isValid ? null : message
  }

  async validateAll(values: Record<string, unknown>): Promise<Record<string, string>> {
    const errors: Record<string, string> = {}

    for (const [fieldName, value] of Object.entries(values)) {
      const error = await this.validateField(fieldName, value)
      if (error) {
        errors[fieldName] = error
      }
    }

    return errors
  }
}
```

### 3.2 表单 Hook（修复闭包过期问题）

```typescript
// hooks/useForm.ts

import { useState, useCallback, useMemo, useRef } from 'react'
import { FormValidator, FormConfig, FieldConfig } from '@/utils/formValidation'

interface UseFormReturn<T extends Record<string, unknown>> {
  values: T
  errors: Record<string, string>
  touched: Record<string, boolean>
  loading: boolean
  setValue: (field: keyof T, value: T[keyof T]) => void
  setValues: (values: Partial<T>) => void
  setFieldError: (field: keyof T, error: string | null) => void
  handleBlur: (field: keyof T) => void
  handleChange: (field: keyof T) => (e: React.ChangeEvent<HTMLElement>) => void
  validate: () => Promise<boolean>
  validateField: (field: keyof T) => Promise<string | null>
  reset: () => void
  submit: (onSubmit: (values: T) => Promise<void>) => Promise<void>
}

interface useFormProps<T> {
  initialValues: T;
  config: FormConfig;
  onSubmit: (values: T) => Promise<void>;
  onValidate?: (errors: Record<string, string>) => void;
}

export function useForm<T extends Record<string, unknown>>(
  config: FormConfig,
  initialValues: T
): UseFormReturn<T> {
  const [values, setValuesState] = useState<T>(initialValues)
  const [errors, setErrors] = useState<Record<string, string>>({})
  const [touched, setTouched] = useState<Record<string, boolean>>({})
  const [loading, setLoading] = useState(false)

  // 【修复 1】使用 useRef 缓存 config 引用，避免 config 变化时 validator 重复创建
  const configRef = useRef(config)
  configRef.current = config

  const validator = useMemo(() => new FormValidator(config), [config])

  // 【修复 2】使用函数式更新 + 字段名作为依赖，避免 config.fields 闭包过期
  const setValue = useCallback((field: keyof T, value: T[keyof T]) => {
    setValuesState(prev => {
      const newValues = { ...prev, [field]: value }
      
      // 在新值计算后再进行验证，确保使用最新的 values
      const fieldConfig = configRef.current.fields[field as string]
      if (fieldConfig?.rules?.some(r => r.trigger === 'onChange') && touched[field as string]) {
        validator.validateField(field as string, value).then((error) => {
          setErrors(prevErrors => ({
            ...prevErrors,
            [field as string]: error || ''
          }))
        })
      }
      
      return newValues
    })
  }, [touched, validator]) // 移除 config.fields 依赖，使用 configRef 代替

  const setValues = useCallback((newValues: Partial<T>) => {
    setValuesState(prev => ({ ...prev, ...newValues }))
  }, [])

  const setFieldError = useCallback((field: keyof T, error: string | null) => {
    setErrors(prev => ({
      ...prev,
      [field as string]: error || ''
    }))
  }, [])

  // 【修复 3】handleBlur 同样使用 configRef 避免闭包过期
  const handleBlur = useCallback(async (field: keyof T) => {
    setTouched(prev => ({ ...prev, [field as string]: true }))

    const fieldConfig = configRef.current.fields[field as string]
    if (fieldConfig?.rules?.some(r => r.trigger === 'onBlur')) {
      const error = await validator.validateField(field as string, values[field as string])
      setErrors(prev => ({
        ...prev,
        [field as string]: error || ''
      }))
    }
  }, [values, validator]) // 移除 config.fields 依赖

  const handleChange = useCallback((field: keyof T) => {
    return (e: React.ChangeEvent<HTMLElement>) => {
      const target = e.target as HTMLInputElement | HTMLTextAreaElement | HTMLSelectElement
      setValue(field, target.value as T[keyof T])
    }
  }, [setValue])

  // 【修复 4】validate 使用 configRef 获取最新字段列表
  const validate = useCallback(async (): Promise<boolean> => {
    const allErrors = await validator.validateAll(values)
    setErrors(allErrors)

    // 标记所有字段为已触碰
    const allTouched: Record<string, boolean> = {}
    Object.keys(configRef.current.fields).forEach(field => {
      allTouched[field] = true
    })
    setTouched(allTouched)

    return Object.keys(allErrors).length === 0
  }, [values, validator]) // 移除 config.fields 依赖

  const validateField = useCallback(async (field: keyof T): Promise<string | null> => {
    return validator.validateField(field as string, values[field as string])
  }, [values, validator])

  const reset = useCallback(() => {
    setValuesState(initialValues)
    setErrors({})
    setTouched({})
  }, [initialValues])

  const submit = useCallback(async (onSubmit: (values: T) => Promise<void>): Promise<void> => {
    const isValid = await validate()
    if (!isValid) return

    setLoading(true)
    try {
      await onSubmit(values)
    } finally {
      setLoading(false)
    }
  }, [validate, values])

  return {
    values,
    errors,
    touched,
    loading,
    setValue,
    setValues,
    setFieldError,
    handleBlur,
    handleChange,
    validate,
    validateField,
    reset,
    submit
  }
}
```

```typescript
// hooks/useVirtualList.ts

import { useState, useRef, useEffect, useCallback, useMemo } from 'react'

export interface VirtualListOptions<T> {
  data: T[]
  itemHeight: number
  overscan?: number // 预渲染数量
  containerRef?: React.RefObject<HTMLElement>
}

export interface VirtualListReturn<T> {
  visibleData: T[]
  startIndex: number
  endIndex: number
  totalHeight: number
  offsetTop: number
  handleScroll: () => void
  scrollToIndex: (index: number) => void
  scrollToTop: () => void
  scrollToBottom: () => void
}

export function useVirtualList<T>(options: VirtualListOptions<T>): VirtualListReturn<T> {
  const { data, itemHeight, overscan = 5, containerRef } = options

  const [scrollTop, setScrollTop] = useState(0)
  const [containerHeight, setContainerHeight] = useState(0)
  const internalRef = useRef<HTMLDivElement>(null)
  const rafRef = useRef<number | null>(null)

  const container = containerRef?.current || internalRef.current

  // 计算可视区域
  const calculateVisibleRange = useCallback(() => {
    if (!container || itemHeight <= 0) return { start: 0, end: 0 }

    const scrollTop = container.scrollTop
    const containerHeight = container.clientHeight

    const startIndex = Math.floor(scrollTop / itemHeight)
    const visibleCount = Math.ceil(containerHeight / itemHeight)
    const endIndex = Math.min(startIndex + visibleCount, data.length)

    return {
      start: Math.max(0, startIndex - overscan),
      end: Math.min(data.length, endIndex + overscan)
    }
  }, [container, itemHeight, data.length, overscan, scrollTop, containerHeight])

  // 处理滚动
  const handleScroll = useCallback(() => {
    if (!container) return

    // 使用 requestAnimationFrame 节流
    if (rafRef.current) {
      cancelAnimationFrame(rafRef.current)
    }

    rafRef.current = requestAnimationFrame(() => {
      setScrollTop(container.scrollTop)
      setContainerHeight(container.clientHeight)
      rafRef.current = null
    })
  }, [container])

  // 初始化滚动监听
  useEffect(() => {
    if (!container) return

    // 初始计算
    setScrollTop(container.scrollTop)
    setContainerHeight(container.clientHeight)

    container.addEventListener('scroll', handleScroll, { passive: true })

    // ResizeObserver 监听容器大小变化
    const resizeObserver = new ResizeObserver(() => {
      setContainerHeight(container.clientHeight)
    })
    resizeObserver.observe(container)

    return () => {
      container.removeEventListener('scroll', handleScroll)
      resizeObserver.disconnect()
      if (rafRef.current) {
        cancelAnimationFrame(rafRef.current)
      }
    }
  }, [container, handleScroll])

  const { start, end } = useMemo(() => calculateVisibleRange(), [calculateVisibleRange])

  const visibleData = useMemo(() => data.slice(start, end), [data, start, end])

  const totalHeight = data.length * itemHeight
  const offsetTop = start * itemHeight

  // 滚动到指定索引
  const scrollToIndex = useCallback((index: number) => {
    if (!container || index < 0 || index >= data.length) return
    container.scrollTo({
      top: index * itemHeight,
      behavior: 'smooth'
    })
  }, [container, data.length, itemHeight])

  const scrollToTop = useCallback(() => {
    container?.scrollTo({ top: 0, behavior: 'smooth' })
  }, [container])

  const scrollToBottom = useCallback(() => {
    if (!container) return
    container.scrollTo({
      top: totalHeight - container.clientHeight,
      behavior: 'smooth'
    })
  }, [container, totalHeight])

  return {
    visibleData,
    startIndex: start,
    endIndex: end,
    totalHeight,
    offsetTop,
    handleScroll,
    scrollToIndex,
    scrollToTop,
    scrollToBottom
  }
}
```

```typescript
  })
```

### 5.1 操作日志与乐观更新

```typescript
// stores/withOptimisticUpdates.ts

import { StateCreator } from 'zustand'

// 操作日志条目
export interface OperationLog<T = unknown> {
  id: string
  type: string
  prevState: T
  newState: T
  timestamp: number
  status: 'pending' | 'success' | 'error'
  rollback?: () => void
}

// 支持乐观更新的 Store 配置
export interface OptimisticStoreConfig<T> {
  maxLogSize?: number // 最大日志数量
  onRollback?: (log: OperationLog<T>) => void
}

// 为 Zustand Store 添加乐观更新能力
export function withOptimisticUpdates<T extends object>(
  creator: StateCreator<T>,
  config: OptimisticStoreConfig<T> = {}
): StateCreator<T & {
  _operationLogs: OperationLog<T>[]
  _pushLog: (log: OperationLog<T>) => void
  _rollback: (logId: string) => void
  optimisticUpdate: <K extends keyof T>(
    key: K,
    newValue: T[K],
    apiCall: () => Promise<void>
  ) => Promise<boolean>
}> {
  const { maxLogSize = 50, onRollback } = config

  return (set, get, api) => ({
    ...creator(set, get, api),
    _operationLogs: [],

    _pushLog: (log: OperationLog<T>) => {
      const current = get()._operationLogs
      const newLogs = [log, ...current].slice(0, maxLogSize)
      set({ _operationLogs: newLogs })
    },

    _rollback: (logId: string) => {
      const log = get()._operationLogs.find(l => l.id === logId)
      if (!log) return

      // 恢复之前的状态
      const prevState = log.prevState
      set({ ...prevState })

      // 调用回调
      onRollback?.(log)

      // 更新日志状态
      const logs = get()._operationLogs.map(l =>
        l.id === logId ? { ...l, status: 'rolled_back' as const } : l
      )
      set({ _operationLogs: logs })
    },

    optimisticUpdate: async (key, newValue, apiCall) => {
      const currentState = get()
      const prevState = { ...currentState } as T

      // 1. 立即更新状态（乐观更新）
      set({ [key]: newValue } as Partial<T>)

      // 2. 记录操作日志
      const log: OperationLog<T> = {
        id: `op_${Date.now()}_${Math.random().toString(36).slice(2)}`,
        type: 'optimistic_update',
        prevState,
        newState: { ...currentState, [key]: newValue } as T,
        timestamp: Date.now(),
        status: 'pending',
        rollback: () => {
          set({ ...prevState })
        }
      }
      get()._pushLog(log)

      try {
        // 3. 执行 API 调用
        await apiCall()

        // 4. API 成功，更新日志状态
        const logs = get()._operationLogs.map(l =>
          l.id === log.id ? { ...l, status: 'success' as const } : l
        )
        set({ _operationLogs: logs })

        return true
      } catch (error) {
        // 5. API 失败，回滚
        get()._rollback(log.id)
        console.error('[Optimistic Update Failed]:', error)
        return false
      }
    }
  })
}
```

```typescript
// stores/notificationStore.ts

import { create } from 'zustand'
import { persist } from 'zustand/middleware'

export interface Notification {
  id: string

  message: string
  createdAt: string
  read: boolean
  actionUrl?: string

interface NotificationState {

  unreadCount: number
  readIds: Set<string>
  fetchNotifications: () => Promise<void>
  markAsRead: (id: string) => Promise<void>
  markAllAsRead: () => Promise<void>
  deleteNotification: (id: string) => Promise<void>
  clearAll: () => void
}

const STORAGE_KEY = 'orion_read_notification_ids'

// 从 localStorage 读取已读 ID 集合
function loadReadIds(): Set<string> {
  try {
    const stored = localStorage.getItem(STORAGE_KEY)
    return new Set(stored ? JSON.parse(stored) : [])
  } catch {
    return new Set()
  }
}

// 保存已读 ID 集合到 localStorage
function saveReadIds(ids: Set<string>): void {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(Array.from(ids)))
}

export const useNotificationStore = create<NotificationState>()(
  persist(
    (set, get) => ({
      notifications: [],
      unreadCount: 0,
      readIds: loadReadIds(),

      fetchNotifications: async () => {
        const response = await fetch('/api/notifications')
        const notifications: Notification[] = await response.json()
        const readIds = get().readIds

        // 本地标记已读状态
        const enriched = notifications.map(n => ({
          ...n,
          read: readIds.has(n.id)
        }))

        set({
          notifications: enriched,
          unreadCount: enriched.filter(n => !n.read).length
        })
      },

      markAsRead: async (id: string) => {
        const { readIds, notifications } = get()

        // 更新本地状态
        const newReadIds = new Set(readIds)
        newReadIds.add(id)
        saveReadIds(newReadIds)

        const newNotifications = notifications.map(n =>
          n.id === id ? { ...n, read: true } : n
        )

        set({
          readIds: newReadIds,
          notifications: newNotifications,
          unreadCount: newNotifications.filter(n => !n.read).length
        })

        // 同步到其他标签页
        tabSyncManager.syncNotification({
          notificationId: id,
          action: 'read'
        })
      },

      markAllAsRead: async () => {
        const { notifications } = get()
        const newReadIds = new Set(notifications.map(n => n.id))
        saveReadIds(newReadIds)

        const newNotifications = notifications.map(n => ({ ...n, read: true }))

        set({
          readIds: newReadIds,
          notifications: newNotifications,
          unreadCount: 0
        })
      },

      deleteNotification: async (id: string) => {
        const { notifications } = get()
        const newNotifications = notifications.filter(n => n.id !== id)

        set({
          notifications: newNotifications,
          unreadCount: newNotifications.filter(n => !n.read).length
        })
      },

      clearAll: () => {
        set({
          notifications: [],
          unreadCount: 0
        })
      }
    }),
    {
      name: 'notification-storage',
      partialize: (state) => ({ readIds: Array.from(state.readIds) }),
      // 自定义序列化/反序列化
      serialize: (state) => JSON.stringify({ readIds: Array.from(state.readIds) }),
      deserialize: (str) => {
        const parsed = JSON.parse(str)
        return { readIds: new Set(parsed.readIds || []) }
      }
    }
  )
)
```

### 5.3 Pipeline 状态更新队列（带重试机制）

#### 问题分析

原实现存在以下问题：

```typescript
// ❌ 问题代码
processQueue: async () => {
  try {
    await fetch('/api/pipelines/batch-update', { ... })
    set({ updateQueue: [], isProcessing: false })  // 问题：无论成功失败都清空
  } catch (error) {
    set({ isProcessing: false })  // 失败时队列清空，数据丢失
  }
}
```

**问题点：**
1. 失败时没有重试机制
2. 队列清空逻辑不正确，导致数据丢失
3. 没有持久化，页面刷新后丢失
4. 没有网络状态感知

#### 修复方案

设计健壮的更新队列，具备以下特性：
- 失败重试机制（指数退避）
- 失败项目保留在队列
- 最大重试次数限制
- localStorage 持久化
- 网络恢复自动重连
- WebSocket 状态同步

```typescript
  isProcessing: boolean
  isConnected: boolean

  addUpdate: (update: Omit<QueuedUpdate, 'id' | 'createdAt' | 'retryCount' | 'status'>) => void
  processQueue: () => Promise<void>
  retryFailed: () => void
  clearCompleted: () => void
  clearQueue: () => void

  setPipelines: (pipelines: Pipeline[]) => void
  setCurrentRun: (run: Pipeline | null) => void
  updatePipelineLocal: (id: string, updates: Partial<Pipeline>) => void
  optimisticUpdate: (id: string, updates: Partial<Pipeline>) => Promise<void>

  persist: () => void
  hydrate: () => void

  handleWebSocketMessage: (message: PipelineUpdate) => void
}

export interface PipelineUpdate {
  type: 'status' | 'log' | 'progress' | 'sync'
  pipelineId: string
  data: Partial<Pipeline>
  timestamp: number
}

// ==================== 常量配置 ====================

const STORAGE_KEY = 'orion_pipeline_queue'
const MAX_RETRIES = 3
const BASE_DELAY_MS = 1000
const MAX_DELAY_MS = 30000
const BATCH_SIZE = 10
const RECONNECT_DELAY_MS = 5000

// ==================== 工具函数 ====================

/**
 * 指数退避计算
 * 公式：delay = baseDelay * 2^retryCount + jitter
 */
function calculateBackoff(retryCount: number): number {
  const exponentialDelay = BASE_DELAY_MS * Math.pow(2, retryCount)
  const jitter = Math.random() * 0.3 * exponentialDelay
  return Math.min(exponentialDelay + jitter, MAX_DELAY_MS)
}

function generateId(): string {
  return `upd_${Date.now()}_${Math.random().toString(36).slice(2, 9)}`
}

// ==================== Store 实现 ====================

export const usePipelineStore = create<PipelineState>()(
  persist(
    (set, get) => ({
      pipelines: [],
      currentRun: null,
      updateQueue: [],
      isProcessing: false,
      isConnected: false,

      // ========== 队列操作 ==========

      addUpdate: (update) => {
        const queuedUpdate: QueuedUpdate = {
          ...update,
          id: generateId(),
          retryCount: 0,
          createdAt: Date.now(),
          status: 'pending',
          maxRetries: MAX_RETRIES
        }

        set((state) => ({
          updateQueue: [...state.updateQueue, queuedUpdate]
        }))

        get().processQueue()
      },

      processQueue: async () => {
        const { updateQueue, isProcessing } = get()

        if (isProcessing) return

        const pendingUpdates = updateQueue.filter(
          u => u.status === 'pending' || 
               (u.status === 'error' && u.retryCount < u.maxRetries)
        )

        if (pendingUpdates.length === 0) return

        set({ isProcessing: true })

        const batches = Math.ceil(pendingUpdates.length / BATCH_SIZE)
        
        for (let i = 0; i < batches; i++) {
          const batch = pendingUpdates.slice(
            i * BATCH_SIZE,
            (i + 1) * BATCH_SIZE
          )

          const results = await Promise.allSettled(
            batch.map(update => executeUpdate(update))
          )

          set((state) => {
            const newQueue = [...state.updateQueue]
            
            results.forEach((result, index) => {
              const update = batch[index]
              const updateIndex = newQueue.findIndex(u => u.id === update.id)
              
              if (updateIndex === -1) return

              if (result.status === 'fulfilled') {
                // ✅ 成功
                newQueue[updateIndex] = {
                  ...update,
                  status: 'success',
                  lastAttemptAt: Date.now()
                }
              } else {
                // ❌ 失败
                const newRetryCount = update.retryCount + 1
                const isExhausted = newRetryCount >= update.maxRetries

                newQueue[updateIndex] = {
                  ...update,
                  retryCount: newRetryCount,
                  status: isExhausted ? 'exhausted' : 'error',
                  error: result.reason?.message || 'Unknown error',
                  lastAttemptAt: Date.now()
                }

                if (!isExhausted) {
                  const delay = calculateBackoff(newRetryCount)
                  setTimeout(() => {
                    set((s) => {
                      const q = [...s.updateQueue]
                      const idx = q.findIndex(u => u.id === update.id)
                      if (idx !== -1 && q[idx].status === 'error') {
                        q[idx] = { ...q[idx], status: 'pending' }
                      }
                      return { updateQueue: q }
                    })
                  }, delay)
                }
              }
            })

            return { updateQueue: newQueue }
          })
        }

        set({ isProcessing: false })
        get().persist()
      },

      retryFailed: () => {
        set((state) => {
          const newQueue = state.updateQueue.map(u => {
            if (u.status === 'error' || u.status === 'exhausted') {
              return { ...u, status: 'pending' as const, retryCount: 0 }
            }
            return u
          })
          return { updateQueue: newQueue }
        })
        get().processQueue()
      },

      clearCompleted: () => {
        set((state) => ({
          updateQueue: state.updateQueue.filter(
            u => u.status !== 'success' && u.status !== 'exhausted'
          )
        }))
      },

      clearQueue: () => {
        set({ updateQueue: [], isProcessing: false })
        localStorage.removeItem(STORAGE_KEY)
      },

      // ========== 状态更新 ==========

      setPipelines: (pipelines) => set({ pipelines }),

      setCurrentRun: (run) => set({ currentRun: run }),

      updatePipelineLocal: (id, updates) => {
        set((state) => {
          const newPipelines = state.pipelines.map(p =>
            p.id === id ? { ...p, ...updates } : p
          )
          const newCurrentRun = state.currentRun?.id === id
            ? { ...state.currentRun, ...updates }
            : state.currentRun

          return { pipelines: newPipelines, currentRun: newCurrentRun }
        })
      },

      optimisticUpdate: async (id, updates) => {
        get().updatePipelineLocal(id, updates)

        let type: UpdateType = 'status'
        if (updates.progress !== undefined) type = 'progress'
        if (updates.currentStage !== undefined) type = 'log'

        get().addUpdate({
          pipelineId: id,
          type,
          payload: updates,
          maxRetries: MAX_RETRIES
        })

        tabSyncManager.syncPipeline({
          pipelineId: id,
          status: updates.status || ''
        })
      },

      // ========== 持久化 ==========

      persist: () => {
        const { updateQueue } = get()
        try {
          localStorage.setItem(STORAGE_KEY, JSON.stringify(
            updateQueue.filter(u => u.status === 'pending' || u.status === 'error')
          ))
        } catch (error) {
          console.error('[PipelineStore] Failed to persist queue:', error)
        }
      },

      hydrate: () => {
        try {
          const stored = localStorage.getItem(STORAGE_KEY)
          if (stored) {
            const pendingUpdates = JSON.parse(stored)
            set({ updateQueue: pendingUpdates })
            setTimeout(() => get().processQueue(), 100)
          }
        } catch (error) {
          console.error('[PipelineStore] Failed to hydrate queue:', error)
        }
      },

      // ========== WebSocket 集成 ==========

      handleWebSocketMessage: (message: PipelineUpdate) => {
        const { updateQueue } = get()

        switch (message.type) {
          case 'sync':
            if (message.data) {
              set((state) => {
                const existingIndex = state.pipelines.findIndex(
                  p => p.id === message.pipelineId
                )

                if (existingIndex !== -1) {
                  const newPipelines = [...state.pipelines]
                  newPipelines[existingIndex] = {
                    ...newPipelines[existingIndex],
                    ...message.data
                  }
                  return { pipelines: newPipelines }
                }

                return {
                  pipelines: [...state.pipelines, message.data as Pipeline]
                }
              })
            }
            break

          case 'status':
          case 'progress':
          case 'log':
            get().updatePipelineLocal(message.pipelineId, message.data)

            const pendingUpdate = updateQueue.find(
              u => u.pipelineId === message.pipelineId && u.status === 'pending'
            )

            if (pendingUpdate) {
              set((state) => ({
                updateQueue: state.updateQueue.map(u =>
                  u.id === pendingUpdate.id ? { ...u, status: 'success' } : u
                )
              }))
            }
            break
        }

        set({ isConnected: true })
      }
    }),

    {
      name: 'pipeline-store',
      partialize: (state) => ({
        updateQueue: state.updateQueue.filter(
          u => u.status === 'pending' || u.status === 'error'
        )
      })
    }
  )
)

// ==================== 辅助函数 ====================

async function executeUpdate(update: QueuedUpdate): Promise<void> {
  const response = await fetch(`/api/pipelines/${update.pipelineId}`, {
    method: 'PATCH',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(update.payload),
    signal: AbortSignal.timeout(10000)
  })

  if (!response.ok) {
    throw new Error(`HTTP ${response.status}: ${response.statusText}`)
  }

  const result = await response.json()
  
  if (result.data) {
    usePipelineStore.getState().updatePipelineLocal(update.pipelineId, result.data)
  }
}

// ==================== 网络状态监听 ====================

export function setupNetworkListener(): () => void {
  const handleOnline = () => {
    console.log('[PipelineStore] Network restored, retrying failed updates')
    setTimeout(() => {
      usePipelineStore.getState().retryFailed()
    }, RECONNECT_DELAY_MS)
  }

  const handleOffline = () => {
    console.log('[PipelineStore] Network offline, queuing updates')
    usePipelineStore.getState().persist()
  }

  window.addEventListener('online', handleOnline)
  window.addEventListener('offline', handleOffline)

  return () => {
    window.removeEventListener('online', handleOnline)
    window.removeEventListener('offline', handleOffline)
  }
}
```

```typescript
function PipelineWebSocket({ pipelineId }: { pipelineId: string }) {
  const handleWebSocketMessage = usePipelineStore(s => s.handleWebSocketMessage)

  useEffect(() => {
    const ws = new WebSocket(`wss://api.example.com/pipelines/${pipelineId}`)

    ws.onmessage = (event) => {
      const message: PipelineUpdate = JSON.parse(event.data)
      handleWebSocketMessage(message)
    }

    ws.onclose = () => {
      usePipelineStore.getState().persist()
    }

    return () => ws.close()
  }, [pipelineId, handleWebSocketMessage])

  return null
}
```

### 6.3.1 闭包过期问题详解

**问题原因：**
```typescript
// ❌ 错误写法
const setValue = useCallback((field: string, value: any) => {
  const fieldConfig = config.fields[field];  // 闭包捕获的是旧 config.fields
}, [config.fields, touched, validator]);
```

当 `config` 对象引用变化时（例如父组件重新渲染传入新对象），`config.fields` 的旧引用仍被闭包捕获，导致验证逻辑使用过期配置。

**解决方案：**
1. 使用 `useRef` 缓存 `config` 引用
2. 在回调中通过 `configRef.current` 访问最新值
3. 从依赖数组中移除 `config.fields`，仅保留 `validator`

```typescript
// ✅ 正确写法
const configRef = useRef(config);
configRef.current = config;

const setValue = useCallback((field: keyof T, value: T[keyof T]) => {
  setValuesState(prev => {
    const fieldConfig = configRef.current.fields[field];  // 总是访问最新值
    // ...
  });
}, [touched, validator]);  // 移除 config.fields 依赖
```
