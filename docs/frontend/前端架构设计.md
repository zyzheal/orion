### 2.1 Zustand Store 设计

```typescript
// 认证状态管理
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

interface User {
  id: string;
  name: string;
  email: string;
  role: string;
  teamIds: string[];
}

interface AuthState {
  user: User | null;
  token: string | null;
  isAuthenticated: boolean;
  login: (user: User, token: string) => void;
  logout: () => void;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set) => ({
      user: null,
      token: null,
      isAuthenticated: false,
      
      login: (user, token) => set({ 
        user, 
        token, 
        isAuthenticated: true 
      }),
      
      logout: () => set({ 
        user: null, 
        token: null, 
        isAuthenticated: false 
      }),
    }),
    {
      name: 'auth-storage',
      partialize: (state) => ({ token: state.token }),
    }
  )
);

// 流水线状态管理
interface PipelineRun {
  id: string;
  pipelineId: string;
  status: 'pending' | 'running' | 'success' | 'failed' | 'cancelled';
  stages: Stage[];
  startTime: string;
  endTime?: string;
}

interface Stage {
  name: string;
  status: 'pending' | 'running' | 'success' | 'failed' | 'skipped';
  duration?: number;
  logs?: string[];
}

interface PipelineState {
  runs: PipelineRun[];
  currentRun: PipelineRun | null;
  loading: boolean;
  
  fetchRuns: (pipelineId: string) => Promise<void>;
  fetchRunDetail: (runId: string) => Promise<void>;
  updateRun: (run: PipelineRun) => void;
  appendLog: (runId: string, stageName: string, log: string) => void;
}

export const usePipelineStore = create<PipelineState>()((set) => ({
  runs: [],
  currentRun: null,
  loading: false,
  
  fetchRuns: async (pipelineId) => {
    set({ loading: true });
    try {
      const runs = await api.getPipelineRuns(pipelineId);
      set({ runs, loading: false });
    } catch (error) {
      set({ loading: false });
      throw error;
    }
  },
  
  fetchRunDetail: async (runId) => {
    set({ loading: true });
    try {
      const run = await api.getPipelineRun(runId);
      set({ currentRun: run, loading: false });
    } catch (error) {
      set({ loading: false });
      throw error;
    }
  },
  
  updateRun: (run) => set((state) => ({
    currentRun: state.currentRun?.id === run.id ? run : state.currentRun,
    runs: state.runs.map(r => r.id === run.id ? run : r),
  })),
  
  appendLog: (runId, stageName, log) => set((state) => ({
    currentRun: state.currentRun?.id === runId 
      ? {
          ...state.currentRun,
          stages: state.currentRun.stages.map(s => 
            s.name === stageName 
              ? { ...s, logs: [...(s.logs || []), log] }
              : s
          ),
        }
      : state.currentRun,
  })),
}));
```

### 2.2 WebSocket Hook 设计

```typescript
// useWebSocket Hook
import { useEffect, useRef, useCallback } from 'react';
import { usePipelineStore } from '@/stores/pipelineStore';

interface UseWebSocketOptions {
  url: string;
  channels: string[];
  enabled?: boolean;
  reconnectInterval?: number;
  maxReconnectAttempts?: number;
}

export function useWebSocket({
  url,
  channels,
  enabled = true,
  reconnectInterval = 3000,
  maxReconnectAttempts = 5,
}: UseWebSocketOptions) {
  const wsRef = useRef<WebSocket | null>(null);
  const reconnectCountRef = useRef(0);
  const updateRun = usePipelineStore((state) => state.updateRun);
  const appendLog = usePipelineStore((state) => state.appendLog);
  
  const handleMessage = useCallback((event: MessageEvent) => {
    const message = JSON.parse(event.data);
    
    switch (message.type) {
      case 'pipeline_update':
        updateRun(message.data);
        break;
      case 'stage_log':
        appendLog(message.data.runId, message.data.stageName, message.data.log);
        break;
      case 'notification':
        // 处理通知
        break;
      case 'pong':
        // 心跳响应
        break;
    }
  }, [updateRun, appendLog]);
  
  const connect = useCallback(() => {
    if (!enabled) return;
    
    const ws = new WebSocket(url);
    wsRef.current = ws;
    
    ws.onopen = () => {
      reconnectCountRef.current = 0;
      // 订阅频道
      channels.forEach(channel => {
        ws.send(JSON.stringify({ type: 'subscribe', channel }));
      });
      // 启动心跳
      startHeartbeat();
    };
    
    ws.onmessage = handleMessage;
    
    ws.onclose = () => {
      if (reconnectCountRef.current < maxReconnectAttempts) {
        reconnectCountRef.current++;
        setTimeout(connect, reconnectInterval);
      }
    };
    
    ws.onerror = (error) => {
      console.error('WebSocket error:', error);
    };
  }, [url, channels, enabled, handleMessage]);
  
  const disconnect = useCallback(() => {
    if (wsRef.current) {
      wsRef.current.close();
      wsRef.current = null;
    }
  }, []);
  
  useEffect(() => {
    connect();
    return () => disconnect();
  }, [connect, disconnect]);
  
  return { connect, disconnect };
}

// 使用示例
function PipelineDetail({ runId }: { runId: string }) {
  useWebSocket({
    url: 'wss://orion.internal/ws',
    channels: [
      `pipeline:run:${runId}`,
      `stage:${runId}:scan`,
      `stage:${runId}:build`,
    ],
  });
  
  // ...
}
```

### 3.2 断线重连策略

```typescript
class ReconnectStrategy {
  private attempts = 0;
  private maxAttempts = 10;
  private baseDelay = 1000;
  private maxDelay = 30000;
  
  getNextDelay(): number {
    if (this.attempts >= this.maxAttempts) {
      return -1; // 不再重连
    }
    
    // 指数退避
    const delay = Math.min(
      this.baseDelay * Math.pow(2, this.attempts),
      this.maxDelay
    );
    
    // 添加随机抖动
    const jitter = delay * 0.1 * Math.random();
    
    this.attempts++;
    return delay + jitter;
  }
  
  reset() {
    this.attempts = 0;
  }
}
```

```typescript
// API 响应缓存
const cache = new Map<string, { data: any; timestamp: number }>();
const CACHE_TTL = 5 * 60 * 1000; // 5 分钟

async function fetchWithCache<T>(
  key: string,
  fetcher: () => Promise<T>
): Promise<T> {
  const cached = cache.get(key);
  if (cached && Date.now() - cached.timestamp < CACHE_TTL) {
    return cached.data;
  }
  
  const data = await fetcher();
  cache.set(key, { data, timestamp: Date.now() });
  return data;
}
```

### 7.1 WebSocket 生命周期管理

**问题**：WebSocket 连接内存泄漏、断线重连无限制、心跳检测缺失

**修复方案**：详见 `docs/frontend/前端性能优化设计.md`

```typescript
// WebSocket 完整生命周期管理
class WebSocketManager {
  private ws: WebSocket | null = null;
  private reconnectCount = 0;
  private heartbeatTimer: NodeJS.Timeout | null = null;
  
  connect() {
    // 连接超时 3s，心跳 30s，指数退避重连
    const ws = new WebSocket(this.options.url);
    ws.onopen = () => {
      this.reconnectCount = 0;
      this.startHeartbeat(); // 30s ping-pong
    };
    ws.onclose = () => this.handleDisconnect(); // 指数退避：< 5 次
  }
}
```

**关键指标**：
- 连接超时：3s
- 心跳间隔：30s
- 重连策略：1s/2s/4s/8s/16s（指数退避，最大 30s）
