### 1.3 快速部署 (开发环境)

```bash
#!/bin/bash
# 快速部署脚本 - 适用于开发/测试环境
# 前提条件：kubectl 已配置，有可用的 K8s 集群

set -euo pipefail

ORION_VERSION="v1.0.0"
NAMESPACE="orion-dev"

echo "=== Orion 快速部署 ==="

# Step 1: 创建命名空间
echo "[1/8] 创建命名空间..."
kubectl create namespace ${NAMESPACE} --dry-run=client -o yaml | kubectl apply -f -

# Step 2: 部署数据库 (使用 Kind 或本地 MySQL)
echo "[2/8] 部署数据库..."
kubectl apply -n ${NAMESPACE} -f https://raw.githubusercontent.com/orion-platform/deploy/${ORION_VERSION}/db/mysql-standalone.yaml

# Step 3: 部署事件总线
echo "[3/8] 部署 NATS..."
helm repo add nats https://nats-io.github.io/k8s/helm/charts/
helm upgrade --install nats nats/nats \
    --namespace ${NAMESPACE} \
    --set config.cluster.enabled=false \
    --set config.jetstream.enabled=true

# Step 4: 部署核心服务
echo "[4/8] 部署核心服务..."
kubectl apply -n ${NAMESPACE} -f https://raw.githubusercontent.com/orion-platform/deploy/${ORION_VERSION}/core/

# Step 5: 部署 Tekton
echo "[5/8] 部署 Tekton..."
kubectl apply --filename https://storage.googleapis.com/tekton-releases/pipeline/latest/release.yaml

# Step 6: 部署 ArgoCD
echo "[6/8] 部署 ArgoCD..."
kubectl apply -n ${NAMESPACE} -k https://github.com/argoproj/argo-cd/manifests/install.yaml

# Step 7: 部署 Orion API
echo "[7/8] 部署 Orion API..."
helm repo add orion https://orion-platform.github.io/helm-charts/
helm upgrade --install orion orion/orion \
    --namespace ${NAMESPACE} \
    --values https://raw.githubusercontent.com/orion-platform/deploy/${ORION_VERSION}/values-dev.yaml

# Step 8: 验证部署
echo "[8/8] 验证部署..."
kubectl wait --for=condition=available deployment --all --namespace=${NAMESPACE} --timeout=300s

echo ""
echo "=== 部署完成 ==="
echo "访问地址：$(kubectl get ingress -n ${NAMESPACE} -o jsonpath='{.items[0].spec.rules[0].host}')"
echo "初始密码：admin / admin123"
```

### 1.4 生产环境部署

```bash
#!/bin/bash
# 生产环境部署脚本
# 前提条件：
# - K8s 集群 (多节点，多可用区)
# - RDS MySQL (高可用)
# - ElastiCache Redis (集群模式)
# - S3 存储桶

set -euo pipefail

ORION_VERSION="v1.0.0"
NAMESPACE="orion-prod"
ENV="prod"

# 加载生产环境配置
source ./config/prod.env

echo "=== Orion 生产环境部署 ==="

# Step 1: 创建命名空间和网络策略
echo "[1/10] 创建命名空间..."
kubectl apply -f - <<EOF
apiVersion: v1
kind: Namespace
metadata:
  name: ${NAMESPACE}
  labels:
    pod-security.kubernetes.io/enforce: restricted
    network-policy: isolated
---
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: default-deny
  namespace: ${NAMESPACE}
spec:
  podSelector: {}
  policyTypes:
  - Ingress
  - Egress
EOF

# Step 2: 部署数据库 (RDS - 外部)
echo "[2/10] 配置数据库连接..."
kubectl create secret generic mysql-credentials \
    --namespace ${NAMESPACE} \
    --from-literal=host=${RDS_HOST} \
    --from-literal=port=3306 \
    --from-literal=database=orion_prod \
    --from-literal=username=${DB_USER} \
    --from-literal=password=${DB_PASSWORD}

# Step 3: 部署 Redis (ElastiCache - 外部)
echo "[3/10] 配置 Redis 连接..."
kubectl create secret generic redis-credentials \
    --namespace ${NAMESPACE} \
    --from-literal=host=${REDIS_HOST} \
    --from-literal=port=6379 \
    --from-literal=password=${REDIS_PASSWORD}

# Step 4: 部署 NATS 集群
echo "[4/10] 部署 NATS 集群..."
helm upgrade --install nats nats/nats \
    --namespace ${NAMESPACE} \
    --values values/nats-prod.yaml \
    --set replicaCount=3 \
    --set config.jetstream.fileStorage.size=50Gi

# Step 5: 部署 Tekton (多 Worker)
echo "[5/10] 部署 Tekton..."
kubectl apply -f https://storage.googleapis.com/tekton-releases/pipeline/latest/release.yaml
kubectl apply -f https://storage.googleapis.com/tekton-releases/triggers/latest/release.yaml

# Step 6: 部署 ArgoCD (高可用)
echo "[6/10] 部署 ArgoCD HA..."
kubectl apply -n ${NAMESPACE} -k github.com/argoproj/argo-cd/manifests/ha?ref=v2.9.0

# Step 7: 部署 Argo Rollouts
echo "[7/10] 部署 Argo Rollouts..."
kubectl apply -n ${NAMESPACE} -f https://github.com/argoproj/argo-rollouts/releases/latest/download/install.yaml

# Step 8: 部署 Orion 核心服务
echo "[8/10] 部署 Orion 核心服务..."
helm upgrade --install orion orion/orion \
    --namespace ${NAMESPACE} \
    --values values/orion-prod.yaml \
    --set image.tag=${ORION_VERSION} \
    --set replicaCount.api=3 \
    --set replicaCount.worker=2 \
    --set resources.api.requests.cpu=500m \
    --set resources.api.requests.memory=1Gi \
    --set autoscaling.api.enabled=true \
    --set autoscaling.api.minReplicas=3 \
    --set autoscaling.api.maxReplicas=10

# Step 9: 部署监控
echo "[9/10] 部署监控..."
kubectl apply -n ${NAMESPACE} -f monitoring/prometheus-rules.yaml
kubectl apply -n ${NAMESPACE} -f monitoring/grafana-dashboards.yaml

# Step 10: 部署 Ingress 和 TLS
echo "[10/10] 配置 Ingress 和 TLS..."
kubectl apply -n ${NAMESPACE} -f networking/ingress-prod.yaml
kubectl apply -n ${NAMESPACE} -f networking/certificates.yaml

echo ""
echo "=== 生产环境部署完成 ==="
echo "请执行以下命令验证："
echo "  kubectl get pods -n ${NAMESPACE}"
echo "  kubectl get svc -n ${NAMESPACE}"
```

### 3.2 Prometheus 告警规则

```yaml
apiVersion: monitoring.coreos.com/v1
kind: PrometheusRule
metadata:
  name: orion-alerts
  namespace: orion-prod
spec:
  groups:
  - name: orion-application
    interval: 30s
    rules:
    # API 错误率过高
    - alert: OrionAPIHighErrorRate
      expr: |
        sum(rate(orion_api_requests_total{status=~"5.."}[5m])) 
        / sum(rate(orion_api_requests_total[5m])) > 0.01
      for: 5m
      labels:
        severity: critical
      annotations:
        summary: "Orion API 错误率过高"
        description: "API 错误率 {{ $value | humanizePercentage }} 超过 1%"
    
    # API 延迟过高
    - alert: OrionAPIHighLatency
      expr: |
        histogram_quantile(0.99, 
          rate(orion_api_request_duration_seconds_bucket[5m])
        ) > 2
      for: 10m
      labels:
        severity: warning
      annotations:
        summary: "Orion API 延迟过高"
        description: "API p99 延迟 {{ $value }}s 超过 2s"
    
    # 任务堆积
    - alert: OrionWorkerQueueBacklog
      expr: orion_worker_queue_depth > 100
      for: 10m
      labels:
        severity: warning
      annotations:
        summary: "Orion Worker 任务堆积"
        description: "队列深度 {{ $value }} 超过 100"
    
    # Pipeline 失败
    - alert: TektonPipelineRunFailed
      expr: tekton_pipelinerun_failed == 1
      for: 0m
      labels:
        severity: warning
      annotations:
        summary: "Tekton Pipeline 运行失败"
        description: "Pipeline {{ $labels.pipelinerun }} 失败"
    
    # ArgoCD 同步失败
    - alert: ArgoCDAppNotSynced
      expr: argocd_app_info{sync_status!="Synced"} == 1
      for: 15m
      labels:
        severity: warning
      annotations:
        summary: "ArgoCD 应用未同步"
        description: "应用 {{ $labels.app }} 同步状态 {{ $labels.sync_status }}"
    
    # AI Token 预算超支
    - alert: OrionAITokenBudgetExceeded
      expr: sum(orion_llm_token_usage_total) > 1000000
      for: 0m
      labels:
        severity: warning
      annotations:
        summary: "AI Token 使用超预算"
        description: "当前 Token 使用量 {{ $value }}"

  - name: orion-database
    rules:
    # MySQL 连接数过高
    - alert: MySQLConnectionHigh
      expr: mysql_global_status_threads_connected / mysql_global_variables_max_connections > 0.8
      for: 5m
      labels:
        severity: warning
      annotations:
        summary: "MySQL 连接数使用率过高"
        description: "连接数使用率 {{ $value | humanizePercentage }}"
    
    # MySQL 复制延迟
    - alert: MySQLReplicationLag
      expr: mysql_slave_status_seconds_behind_master > 30
      for: 5m
      labels:
        severity: critical
      annotations:
        summary: "MySQL 复制延迟过高"
        description: "复制延迟 {{ $value }}s"
```

## 五、故障排查

### 5.1 常见故障

**故障 1: API 服务响应慢**

```
现象：
- API 延迟从 100ms 上升到 2s+
- 用户报告页面加载慢

排查步骤：
1. 检查 API Pod 状态
   kubectl top pods -n orion-prod | grep orion-api
   
2. 检查数据库连接
   kubectl exec mysql-0 -n orion-prod -- mysql -e "SHOW PROCESSLIST"
   
3. 检查慢查询
   kubectl exec mysql-0 -n orion-prod -- mysql -e "SELECT * FROM mysql.slow_log ORDER BY start_time DESC LIMIT 10"
   
4. 检查 Redis
   kubectl exec redis-0 -n orion-prod -- redis-cli INFO stats
   
5. 查看应用日志
   kubectl logs deployment/orion-api -n orion-prod --tail=100 | grep -i "slow\|timeout"

可能原因：
- 数据库慢查询
- Redis 连接池耗尽
- 外部依赖响应慢
- GC 停顿

解决方案：
- 优化慢查询 SQL
- 增加 Redis 连接池
- 增加超时和降级
- 调整 JVM 参数
```

**故障 2: Pipeline 大量失败**

```
现象：
- 多个 PipelineRun 状态为 Failed
- TaskRun 长时间 Pending

排查步骤：
1. 查看失败的 PipelineRun
   tkn pipelinerun list -n orion-prod | grep Failed
   
2. 查看失败原因
   tkn pipelinerun describe <run-name> -n orion-prod
   
3. 检查 Tekton Controller
   kubectl logs deployment/tekton-pipelines-controller -n tekton-pipelines
   
4. 检查 Worker 资源
   kubectl top nodes
   
5. 检查 Workspace PVC
   kubectl get pvc -n orion-prod

可能原因：
- 构建资源不足
- 镜像拉取失败
- 外部依赖不可用
- Workspace 空间不足

解决方案：
- 扩容 Runner
- 预拉取基础镜像
- 配置镜像仓库代理
- 清理旧 PVC
```

### 5.2 应急预案

**预案 1: 数据库主节点故障**

```bash
#!/bin/bash
# MySQL 主节点故障应急切换

set -euo pipefail

NAMESPACE="orion-prod"
PRIMARY_POD="mysql-0"
REPLICA_POD="mysql-1"

echo "=== MySQL 故障切换 ==="

# 1. 确认主节点不可达
if kubectl exec ${PRIMARY_POD} -n ${NAMESPACE} -- mysqladmin ping &>/dev/null; then
    echo "主节点仍可访问，无需切换"
    exit 1
fi

echo "主节点不可达，开始切换..."

# 2. 提升从节点为主节点
kubectl exec ${REPLICA_POD} -n ${NAMESPACE} -- mysql <<EOF
STOP REPLICA;
RESET REPLICA ALL;
SET GLOBAL read_only = OFF;
EOF

# 3. 更新服务指向
kubectl patch svc mysql -n ${NAMESPACE} -p '{"spec":{"selector":{"statefulset.kubernetes.io/pod-name":"mysql-1"}}}'

# 4. 重启应用
kubectl rollout restart deployment/orion-api -n ${NAMESPACE}

echo "切换完成，请验证服务正常"
```

### 6.2 恢复流程

```bash
#!/bin/bash
# MySQL 数据恢复流程

set -euo pipefail

BACKUP_FILE=$1
NAMESPACE="orion-prod"

echo "=== MySQL 数据恢复 ==="

# 1. 下载备份
echo "[1/6] 下载备份..."
aws s3 cp s3://orion-backup/mysql/full/${BACKUP_FILE} /tmp/

# 2. 解压备份
echo "[2/6] 解压备份..."
tar -xzf /tmp/${BACKUP_FILE} -C /tmp/

# 3. 停止应用
echo "[3/6] 停止应用..."
kubectl scale deployment/orion-api --replicas=0 -n ${NAMESPACE}
kubectl scale deployment/orion-worker --replicas=0 -n ${NAMESPACE}

# 4. 停止 MySQL
echo "[4/6] 停止 MySQL..."
kubectl exec mysql-0 -n ${NAMESPACE} -- mysqladmin shutdown

# 5. 恢复数据
echo "[5/6] 恢复数据..."
kubectl cp /tmp/${BACKUP_FILE%.tar.gz} mysql-0:/backup/ -n ${NAMESPACE}
kubectl exec mysql-0 -n ${NAMESPACE} -- bash -c "
    xtrabackup --copy-back --target-dir=/backup/${BACKUP_FILE%.tar.gz}
    chown -R mysql:mysql /var/lib/mysql
"

# 6. 启动 MySQL
echo "[6/6] 启动 MySQL..."
kubectl exec mysql-0 -n ${NAMESPACE} -- mysqld_safe &

# 等待 MySQL 就绪
sleep 30
kubectl exec mysql-0 -n ${NAMESPACE} -- mysqladmin ping

# 7. 恢复应用
kubectl scale deployment/orion-api --replicas=3 -n ${NAMESPACE}
kubectl scale deployment/orion-worker --replicas=2 -n ${NAMESPACE}

echo "恢复完成"
```

### 7.2 升级脚本

```bash
#!/bin/bash
# Orion 版本升级脚本

set -euo pipefail

CURRENT_VERSION=$1
TARGET_VERSION=$2
NAMESPACE="orion-prod"

echo "=== Orion 版本升级 ==="
echo "当前版本：${CURRENT_VERSION}"
echo "目标版本：${TARGET_VERSION}"

# 1. 备份当前配置
echo "[1/5] 备份当前配置..."
helm get values orion -n ${NAMESPACE} > backup/values-${CURRENT_VERSION}.yaml

# 2. 备份数据库
echo "[2/5] 备份数据库..."
kubectl create job --from=cronjob/mysql-backup mysql-backup-pre-upgrade -n ${NAMESPACE}
kubectl wait --for=condition=complete job/mysql-backup-pre-upgrade -n ${NAMESPACE}

# 3. 升级 Helm Chart
echo "[3/5] 升级 Helm Chart..."
helm upgrade orion orion/orion \
    --namespace ${NAMESPACE} \
    --version ${TARGET_VERSION} \
    --values backup/values-${CURRENT_VERSION}.yaml \
    --timeout 10m

# 4. 等待升级完成
echo "[4/5] 等待升级完成..."
kubectl rollout status deployment/orion-api -n ${NAMESPACE}
kubectl rollout status deployment/orion-worker -n ${NAMESPACE}

# 5. 验证升级
echo "[5/5] 验证升级..."
kubectl exec deployment/orion-api -n ${NAMESPACE} -- curl -s http://localhost:8080/health | jq .

# 检查版本号
VERSION=$(kubectl exec deployment/orion-api -n ${NAMESPACE} -- curl -s http://localhost:8080/api/version | jq -r .version)
if [ "${VERSION}" == "${TARGET_VERSION}" ]; then
    echo "升级成功！当前版本：${VERSION}"
else
    echo "升级失败！期望版本：${TARGET_VERSION}, 实际版本：${VERSION}"
    exit 1
fi
```
