## 一、概述

### 1.1 设计目标

构建缓存是 CI-CD 性能优化的核心能力，本设计旨在提供**灵活可配置**的缓存管理方案，支持：

- ✅ 灵活启用/禁用缓存（全局/流水线/任务级别）
- ✅ 多种缓存模式（自动/自定义/禁用）
- ✅ 多语言依赖缓存智能识别
- ✅ 缓存淘汰策略可配置（LRU/LFU/FIFO）
- ✅ 缓存命中率监控与分析
- ✅ 缓存清理与归档策略

### 1.2 适用场景

| 场景 | 配置级别 | 说明 |
|------|---------|------|
| **全局默认** | 平台级 | 平台默认缓存策略，所有流水线继承 |
| **流水线模板** | 模板级 | 特定类型流水线的缓存配置 |
| **单流水线** | 实例级 | 单个流水线的自定义配置 |
| **单任务** | 任务级 | 特定 Stage/Task 的缓存配置 |

### 1.3 核心价值

```
缓存启用前 vs 启用后对比:

┌─────────────────────────────────────────────────────────────┐
│  指标          │ 无缓存    │ 启用缓存   │ 提升           │
├─────────────────────────────────────────────────────────────┤
│  平均构建时间   │ 8m 30s    │ 2m 15s     │ ↓ 74%          │
│  依赖下载时间   │ 3m 20s    │ 15s        │ ↓ 92%          │
│  构建成功率     │ 92%       │ 97%        │ ↑ 5%           │
│  Runner 资源占用 │ 100%      │ 65%        │ ↓ 35%          │
│  月度构建成本   │ $10,000   │ $6,500     │ ↓ 35%          │
└─────────────────────────────────────────────────────────────┘
```

## 二、缓存配置 Schema

### 2.1 完整配置结构

```yaml
# 构建缓存配置 Schema (orion-pipeline.yaml)
apiVersion: pipeline.orion.dev/v1alpha1
kind: CacheConfig
metadata:
  name: build-cache-default
  namespace: orion-system

spec:
  # ========== 全局开关 ==========
  enabled: true                        # 灵活配置是否启用缓存
  mode: "auto"                         # auto | custom | disabled
  
  # ========== 层缓存配置 ==========
  layerCache:
    enabled: true
    maxLayers: 100                     # 最大缓存层数
    ttl: "7d"                          # 缓存存活时间 (ISO 8601)
    compression: "zstd"                # zstd | gzip | none
    storageClass: "standard"           # standard | fast | archive
  
  # ========== 依赖缓存配置 ==========
  dependencyCache:
    enabled: true
    
    # 按语言配置缓存策略
    languages:
      nodejs:
        enabled: true
        paths:
          - /root/.npm
          - /app/node_modules
        hashFiles:
          - package-lock.json
          - yarn.lock
          - pnpm-lock.yaml
        cachePrefix: "npm"
      
      python:
        enabled: true
        paths:
          - /root/.cache/pip
          - .venv
          - /usr/local/lib/python*/site-packages
        hashFiles:
          - requirements.txt
          - Pipfile.lock
          - poetry.lock
        cachePrefix: "pip"
      
      go:
        enabled: true
        paths:
          - /go/pkg/mod
          - /go/build/cache
        hashFiles:
          - go.mod
          - go.sum
        cachePrefix: "go"
      
      java:
        enabled: true
        paths:
          - /root/.m2
          - /root/.gradle/caches
        hashFiles:
          - pom.xml
          - build.gradle
          - settings.gradle
        cachePrefix: "maven"
      
      rust:
        enabled: true
        paths:
          - /root/.cargo/registry
          - target/debug/.fingerprint
        hashFiles:
          - Cargo.lock
          - Cargo.toml
        cachePrefix: "cargo"
    
    # 自定义缓存路径
    customPaths:
      - name: "build-artifacts"
        paths:
          - ./dist
          - ./build
        hashFiles:
          - Makefile
          - CMakeLists.txt
  
  # ========== 缓存键配置 ==========
  cacheKey:
    strategy: "hash"                   # hash | timestamp | branch
    prefix: "${pipeline.name}"         # 支持变量
    suffix: "${branch.name}"           # 支持变量
    includeFiles:                      # 影响缓存键的文件
      - "**/package-lock.json"
      - "**/requirements.txt"
      - "**/go.mod"
    excludeFiles:                      # 忽略的文件
      - "**/*.md"
      - "**/*.log"
  
  # ========== 缓存淘汰策略 ==========
  eviction:
    policy: "lru"                      # lru | lfu | fifo | ttl
    maxSize: "10GB"                    # 最大缓存大小
    maxAge: "30d"                      # 最大缓存年龄
    minFreeSpace: "5GB"                # 最小可用空间
    
    # LRU 配置
    lru:
      accessThreshold: 3               # 最少访问次数
      timeThreshold: "7d"              # 最少访问时间
    
    # LFU 配置
    lfu:
      frequencyThreshold: 5            # 最少使用频率
      decayFactor: 0.9                 # 衰减因子
    
    # FIFO 配置
    fifo:
      maxItems: 1000                   # 最大缓存项数量
  
  # ========== 缓存清理策略 ==========
  cleanup:
    enabled: true
    schedule: "0 2 * * 0"              # Cron 表达式 (每周日凌晨 2 点)
    keepLastN: 50                      # 保留最近 N 个缓存
    keepByBranch:                      # 按分支保留
      main: 20                         # 主分支保留 20 个
      develop: 10                      # 开发分支保留 10 个
      feature: 5                       # 特性分支保留 5 个
    
    # 归档配置
    archive:
      enabled: true
      storageClass: "archive"          # 归档存储类型
      afterDays: "30d"                 # 30 天后归档
      deleteAfter: "90d"               # 90 天后删除
  
  # ========== 缓存监控配置 ==========
  monitoring:
    enabled: true
    metrics:
      - cacheHitRate                   # 缓存命中率
      - cacheMissRate                  # 缓存未命中率
      - cacheSize                      # 缓存大小
      - cacheAge                       # 缓存年龄
      - evictionCount                  # 淘汰次数
      - cleanupCount                   # 清理次数
    
    # 告警配置
    alerts:
      - name: "低缓存命中率"
        metric: cacheHitRate
        condition: "< 30%"             # 命中率低于 30% 告警
        duration: "1h"                 # 持续 1 小时
        severity: "warning"
      
      - name: "缓存空间不足"
        metric: cacheFreeSpace
        condition: "< 1GB"
        severity: "critical"
      
      - name: "缓存淘汰频繁"
        metric: evictionCount
        condition: "> 100/hour"
        severity: "warning"
  
  # ========== 缓存共享配置 ==========
  sharing:
    scope: "team"                      # global | team | pipeline | private
    allowedTeams:                      # 允许共享的团队
      - platform-team
      - backend-team
    allowedPipelines:                  # 允许共享的流水线
      - ci-pipeline
      - release-pipeline
  
  # ========== 高级配置 ==========
  advanced:
    # 并发控制
    concurrency:
      maxConcurrentWrites: 2           # 最大并发写入数
      maxConcurrentReads: 10           # 最大并发读取数
    
    # 重试策略
    retry:
      maxRetries: 3
      initialDelay: "1s"
      maxDelay: "30s"
      multiplier: 2.0
    
    # 超时配置
    timeouts:
      upload: "10m"                    # 上传超时
      download: "5m"                   # 下载超时
      cleanup: "30m"                   # 清理超时
    
    # 压缩配置
    compression:
      enabled: true
      algorithm: "zstd"                # zstd | gzip | lz4
      level: 3                         # 压缩级别 (1-9)
      minSize: "1MB"                   # 最小压缩大小
```

## 五、缓存策略详解

### 5.1 缓存键生成策略

```typescript
interface CacheKeyStrategy {
  // 策略类型
  type: 'hash' | 'timestamp' | 'branch' | 'custom';
  
  // 组成部分
  prefix?: string;        // 前缀 (支持变量：${pipeline.name})
  suffix?: string;        // 后缀 (支持变量：${branch.name})
  
  // 文件哈希
  hashFiles?: string[];   // 影响缓存键的文件 glob 模式
  
  // 动态变量
  variables?: {
    pipeline: string;
    branch: string;
    repo: string;
    os: string;
    arch: string;
  };
}

// 示例：Node.js 项目缓存键
// 输入配置:
{
  strategy: 'hash',
  prefix: '${pipeline.name}-${repo.name}',
  hashFiles: ['package-lock.json', 'yarn.lock'],
  variables: {
    pipeline: 'ci-pipeline',
    repo: 'my-app',
    branch: 'main'
  }
}

// 生成的缓存键:
// ci-pipeline-my-app-abc123def456-main
// (abc123def456 是 package-lock.json 的哈希)
```

### 5.2 缓存淘汰算法实现

```typescript
// LRU (Least Recently Used) 实现
class LRUCache {
  private cache: Map<string, CacheItem>;
  private maxSize: number;
  
  get(key: string): CacheItem | null {
    const item = this.cache.get(key);
    if (item) {
      // 更新访问时间
      item.lastAccess = Date.now();
      item.accessCount++;
      this.cache.delete(key);
      this.cache.set(key, item); // 移到末尾
    }
    return item || null;
  }
  
  put(key: string, value: CacheItem): void {
    if (this.cache.size >= this.maxSize) {
      // 淘汰最久未使用的项
      const oldestKey = this.cache.keys().next().value;
      this.cache.delete(oldestKey);
    }
    value.lastAccess = Date.now();
    this.cache.set(key, value);
  }
}

// LFU (Least Frequently Used) 实现
class LFUCache {
  private cache: Map<string, CacheItem>;
  private frequencyMap: Map<number, Set<string>>;
  private minFrequency: number;
  
  get(key: string): CacheItem | null {
    const item = this.cache.get(key);
    if (item) {
      // 增加使用频率
      this.incrementFrequency(key);
    }
    return item || null;
  }
  
  private incrementFrequency(key: string): void {
    const item = this.cache.get(key)!;
    const oldFreq = item.frequency;
    const newFreq = oldFreq + 1;
    
    // 从旧频率列表移除
    this.frequencyMap.get(oldFreq)!.delete(key);
    if (this.frequencyMap.get(oldFreq)!.size === 0) {
      this.frequencyMap.delete(oldFreq);
      if (this.minFrequency === oldFreq) {
        this.minFrequency = newFreq;
      }
    }
    
    // 添加到新频率列表
    if (!this.frequencyMap.has(newFreq)) {
      this.frequencyMap.set(newFreq, new Set());
    }
    this.frequencyMap.get(newFreq)!.add(key);
    
    item.frequency = newFreq;
  }
}
```

### 5.3 缓存命中率计算

```typescript
interface CacheMetrics {
  hitCount: number;      // 命中次数
  missCount: number;     // 未命中次数
  hitRate: number;       // 命中率
  missRate: number;      // 未命中率
  avgSize: number;       // 平均缓存大小
  avgAge: number;        // 平均缓存年龄
  evictionCount: number; // 淘汰次数
}

class CacheMonitor {
  private metrics: CacheMetrics = {
    hitCount: 0,
    missCount: 0,
    hitRate: 0,
    missRate: 0,
    avgSize: 0,
    avgAge: 0,
    evictionCount: 0
  };
  
  recordHit(): void {
    this.metrics.hitCount++;
    this.updateRates();
  }
  
  recordMiss(): void {
    this.metrics.missCount++;
    this.updateRates();
  }
  
  private updateRates(): void {
    const total = this.metrics.hitCount + this.metrics.missCount;
    this.metrics.hitRate = (this.metrics.hitCount / total) * 100;
    this.metrics.missRate = (this.metrics.missCount / total) * 100;
  }
  
  getMetrics(): CacheMetrics {
    return this.metrics;
  }
  
  // 告警检查
  checkAlerts(): Alert[] {
    const alerts: Alert[] = [];
    
    if (this.metrics.hitRate < 30) {
      alerts.push({
        name: '低缓存命中率',
        severity: 'warning',
        message: `当前命中率 ${this.metrics.hitRate.toFixed(1)}% < 30%`,
        metric: 'hitRate',
        value: this.metrics.hitRate
      });
    }
    
    return alerts;
  }
}
```

### 6.2 缓存优化技巧

```yaml
# 技巧 1: 分层缓存策略
cache:
  dependencyCache:
    customPaths:
      - name: "build-cache"
        paths: [./dist, ./build]
        hashFiles: [Makefile]
        strategy: "branch"       # 按分支隔离
      - name: "tool-cache"
        paths: [/root/.cache]
        hashFiles: []            # 不依赖文件，全局共享
        strategy: "global"

cache:
  conditional:
    - if: "branch == 'main'"    # 主分支使用大缓存
      maxSize: "20GB"
      ttl: "30d"
    - if: "branch.startsWith('feature/')"  # 特性分支使用小缓存
      maxSize: "5GB"
      ttl: "7d"

# 技巧 3: 预热缓存
cache:
  warmup:
    enabled: true
    schedule: "0 6 * * 1"       # 周一早上 6 点预热
    pipelines:                  # 预热的流水线
      - main-ci-pipeline
```
