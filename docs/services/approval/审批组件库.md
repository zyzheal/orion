## 1. StatusBadge 组件

审批状态徽章组件，用于展示审批单的不同状态。

### TypeScript 接口定义

```typescript
import React from 'react';

export type StatusType = 'pending' | 'approved' | 'rejected' | 'processing';
export type SizeType = 'sm' | 'md' | 'lg';

export interface StatusBadgeProps {
  status: StatusType;
  size?: SizeType;
  showLabel?: boolean;
  className?: string;
  style?: React.CSSProperties;
}
```

### 样式变体配置

```typescript
const statusVariants = {
  pending: {
    icon: ClockIcon,
    color: 'var(--color-warning-foreground)',
    bgColor: 'var(--color-warning-background)',
    borderColor: 'var(--color-warning-border)',
  },
  approved: {
    icon: CheckCircleIcon,
    color: 'var(--color-success-foreground)',
    bgColor: 'var(--color-success-background)',
    borderColor: 'var(--color-success-border)',
  },
  rejected: {
    icon: XCircleIcon,
    color: 'var(--color-error-foreground)',
    bgColor: 'var(--color-error-background)',
    borderColor: 'var(--color-error-border)',
  },
  processing: {
    icon: LoadingSpinnerIcon,
    color: 'var(--color-info-foreground)',
    bgColor: 'var(--color-info-background)',
    borderColor: 'var(--color-info-border)',
  },

const sizeVariants = {
  sm: { width: 16, height: 16, fontSize: 'var(--font-size-xs)' },
  md: { width: 20, height: 20, fontSize: 'var(--font-size-sm)' },
  lg: { width: 24, height: 24, fontSize: 'var(--font-size-md)' },
};

const statusLabels = {
  pending: '待审批',
  approved: '已通过',
  rejected: '已拒绝',
  processing: '处理中',
};
```

### 组件实现

```tsx
export const StatusBadge: React.FC<StatusBadgeProps> = ({
  status,
  size = 'md',
  showLabel = true,
  className = '',
  style,
}) => {
  const variant = statusVariants[status];
  const sizes = sizeVariants[size];
  const Icon = variant.icon;

  return (
    <span
      className={`status-badge status-badge--${status} status-badge--${size} ${className}`}
      style={{
        display: 'inline-flex',
        alignItems: 'center',
        gap: 'var(--spacing-1)',
        padding: 'var(--spacing-0\\.5) var(--spacing-2)',
        backgroundColor: variant.bgColor,
        color: variant.color,
        border: `1px solid ${variant.borderColor}`,
        borderRadius: 'var(--border-radius-full)',
        fontSize: sizes.fontSize,
        ...style,
      }}
      role="status"
      aria-label={`当前状态：${statusLabels[status]}`}
      aria-live="polite"
    >
      <Icon
        width={sizes.width}
        height={sizes.height}
        aria-hidden="true"
        className={status === 'processing' ? 'status-badge__spinner' : ''}
      />
      {showLabel && <span>{statusLabels[status]}</span>}
    </span>
  );
};
```

### 使用示例

```tsx
// 基本用法
<StatusBadge status="pending" />

// 仅显示图标
<StatusBadge status="approved" showLabel={false} />

// 大尺寸
<StatusBadge status="rejected" size="lg" />
```

## 2. ActionModal 组件

审批操作弹窗，支持通过、拒绝、转交三种操作类型。

### TypeScript 接口定义

```typescript
import React from 'react';

export type ActionType = 'approve' | 'reject' | 'transfer';

export interface User {
  id: string;
  name: string;
  email: string;
  avatar?: string;
}

export interface ActionModalProps {
  type: ActionType;
  visible: boolean;
  onConfirm: (comment: string, transferTo?: string) => Promise<void>;
  onCancel: () => void;
  loading?: boolean;
  candidateUsers?: User[];
  requiredComment?: boolean;
  placeholder?: string;
}
```

### 表单验证规则

```typescript
const validationRules = {
  approve: {
    requireComment: false,
    maxLength: 500,
  },
  reject: {
    requireComment: true,
    minLength: 10,
    maxLength: 500,
  },
  transfer: {
    requireComment: false,
    requireTargetUser: true,
    maxLength: 200,
  },
};

const modalConfig = {
  approve: {
    title: '确认通过',
    confirmText: '通过',
    confirmVariant: 'primary',
    placeholder: '请输入通过意见（可选）',
  },
  reject: {
    title: '确认拒绝',
    confirmText: '拒绝',
    confirmVariant: 'danger',
    placeholder: '请输入拒绝原因（必填）',
  },
  transfer: {
    title: '转交审批',
    confirmText: '转交',
    confirmVariant: 'primary',
    placeholder: '请输入转交说明（可选）',
  },
};
```

### 组件实现

```tsx
export const ActionModal: React.FC<ActionModalProps> = ({
  type,
  visible,
  onConfirm,
  onCancel,
  loading = false,
  candidateUsers = [],
  requiredComment = false,
  placeholder,
}) => {
  const [comment, setComment] = React.useState('');
  const [transferTo, setTransferTo] = React.useState<string>('');
  const [error, setError] = React.useState<string>('');
  const config = modalConfig[type];
  const rules = validationRules[type];

  const validate = (): boolean => {
    setError('');

    if (type === 'reject' && !comment.trim()) {
      setError('请输入拒绝原因');
      return false;
    }

    if (comment.length > (rules.maxLength || 500)) {
      setError(`评论长度不能超过${rules.maxLength}字`);
      return false;
    }

    if (type === 'transfer' && !transferTo) {
      setError('请选择转交对象');
      return false;
    }

    return true;
  };

  const handleSubmit = async () => {
    if (!validate()) return;
    try {
      await onConfirm(comment, transferTo || undefined);
      setComment('');
      setTransferTo('');
    } catch (e) {
      setError('操作失败，请重试');
    }
  };

  React.useEffect(() => {
    if (!visible) {
      setComment('');
      setTransferTo('');
      setError('');
    }
  }, [visible]);

  return (
    <Modal
      visible={visible}
      title={config.title}
      onCancel={onCancel}
      footer={
        <>
          <Button onClick={onCancel} disabled={loading}>
            取消
          </Button>
          <Button
            onClick={handleSubmit}
            variant={config.confirmVariant}
            loading={loading}
          >
            {config.confirmText}
          </Button>
        </>
      }
      role="dialog"
      aria-modal="true"
      aria-labelledby="modal-title"
    >
      <div className="action-modal__content">
        {type === 'transfer' && candidateUsers.length > 0 && (
          <div className="action-modal__user-selector">
            <label htmlFor="transfer-target">选择接收人</label>
            <UserSelect
              id="transfer-target"
              users={candidateUsers}
              value={transferTo}
              onChange={setTransferTo}
              aria-required="true"
            />
          </div>
        )}

        <div className="action-modal__comment">
          <label htmlFor="comment-input">
            {type === 'reject' && <span className="required">*</span>}
            评论
          </label>
          <TextArea
            id="comment-input"
            value={comment}
            onChange={setComment}
            placeholder={placeholder || config.placeholder}
            maxLength={rules.maxLength}
            rows={4}
            aria-invalid={!!error}
            aria-describedby={error ? 'comment-error' : undefined}
          />
          <div className="action-modal__char-count">
            {comment.length}/{rules.maxLength}
          </div>
        </div>

        {error && (
          <div
            id="comment-error"
            className="action-modal__error"
            role="alert"
          >
            <ErrorIcon aria-hidden="true" />
            {error}
          </div>
        )}
      </div>
    </Modal>
  );
};
```

### 使用示例

```tsx
// 通过操作
<ActionModal
  type="approve"
  visible={showApproveModal}
  onConfirm={async (comment) => {
    await api.approve(requestId, comment);
    setShowApproveModal(false);
  }}
  onCancel={() => setShowApproveModal(false)}
/>

// 转交操作
<ActionModal
  type="transfer"
  visible={showTransferModal}
  candidateUsers={availableApprovers}
  onConfirm={async (comment, transferTo) => {
    await api.transfer(requestId, transferTo!, comment);
    setShowTransferModal(false);
  }}
  onCancel={() => setShowTransferModal(false)}
/>
```

## 3. TimelineItem 组件

时间线条目组件，展示审批流程中的操作记录。

### TypeScript 接口定义

```typescript
export type ActionResult = 'success' | 'failed' | 'skipped';
export type ActionType =
  | 'created'
  | 'approved'
  | 'rejected'
  | 'transferred'
  | 'commented'
  | 'cancelled';

export interface TimelineItemProps {
  timestamp: Date;
  actor: User;
  action: ActionType;
  result?: ActionResult;
  expanded?: boolean;
  details?: React.ReactNode;
  isLast?: boolean;
  onToggle?: () => void;
}
```

### 图标映射

```typescript
const actionIcons = {
  created: PlusCircleIcon,
  approved: CheckCircleIcon,
  rejected: XCircleIcon,
  transferred: ArrowRightCircleIcon,
  commented: MessageCircleIcon,
  cancelled: BanIcon,
};

const resultVariants = {
  success: { color: 'var(--color-success-foreground)', icon: CheckIcon },
  failed: { color: 'var(--color-error-foreground)', icon: XIcon },
  skipped: { color: 'var(--color-neutral-foreground)', icon: MinusIcon },
};
```

### 组件实现

```tsx
export const TimelineItem: React.FC<TimelineItemProps> = ({
  timestamp,
  actor,
  action,
  result = 'success',
  expanded = false,
  details,
  isLast = false,
  onToggle,
}) => {
  const Icon = actionIcons[action];
  const resultVariant = resultVariants[result];
  const hasDetails = Boolean(details);
  const [isExpanded, setIsExpanded] = React.useState(expanded);

  const toggleExpand = () => {
    setIsExpanded(!isExpanded);
    onToggle?.();
  };

  const formatTime = (date: Date) => {
    return new Intl.DateTimeFormat('zh-CN', {
      year: 'numeric',
      month: '2-digit',
      day: '2-digit',
      hour: '2-digit',
      minute: '2-digit',
    }).format(date);
  };

  return (
    <div
      className={`timeline-item ${isLast ? 'timeline-item--last' : ''}`}
      role="listitem"
    >
      {/* 时间线连接器 */}
      <div className="timeline-item__connector" aria-hidden="true">
        <div className="timeline-item__line" />
        <div
          className={`timeline-item__dot timeline-item__dot--${result}`}
          style={{ borderColor: resultVariant.color }}
        >
          <Icon
            width={12}
            height={12}
            color={resultVariant.color}
            aria-hidden="true"
          />
        </div>
      </div>

      {/* 内容区域 */}
      <div className="timeline-item__content">
        <div className="timeline-item__header">
          <Avatar user={actor} size="sm" aria-hidden="true" />
          <div className="timeline-item__meta">
            <span className="timeline-item__actor">{actor.name}</span>
            <span className="timeline-item__action">{getActionText(action)}</span>
            <time
              className="timeline-item__time"
              dateTime={timestamp.toISOString()}
            >
              {formatTime(timestamp)}
            </time>
          </div>
          {result !== 'success' && (
            <resultVariant.icon
              width={16}
              height={16}
              color={resultVariant.color}
              aria-label={`结果：${result}`}
            />
          )}
        </div>

        {/* 可展开的详情 */}
        {hasDetails && (
          <div
            className="timeline-item__details"
            aria-expanded={isExpanded}
          >
            <button
              className="timeline-item__toggle"
              onClick={toggleExpand}
              aria-expanded={isExpanded}
              aria-controls="timeline-details"
            >
              <ChevronIcon
                className={isExpanded ? 'expanded' : ''}
                aria-hidden="true"
              />
              {isExpanded ? '收起详情' : '查看详情'}
            </button>

            {isExpanded && (
              <div id="timeline-details" className="timeline-item__details-content">
                {details}
              </div>
            )}
          </div>
        )}
      </div>
    </div>
  );
};
```

## 4. CommentItem 组件

评论条目组件，支持编辑、删除和 @mentions 功能。

### TypeScript 接口定义

```typescript
export interface CommentItemProps {
  id: string;
  author: User;
  content: string;
  createdAt: Date;
  editedAt?: Date;
  canEdit?: boolean;
  canDelete?: boolean;
  onEdit: (id: string, content: string) => Promise<void>;
  onDelete: (id: string) => Promise<void>;
  onMention?: (userId: string) => void;
  currentUser?: User;
}

export interface MentionUser {
  id: string;
  name: string;
  email: string;
}
```

### 组件实现

```tsx
export const CommentItem: React.FC<CommentItemProps> = ({
  id,
  author,
  content,
  createdAt,
  editedAt,
  canEdit = false,
  canDelete = false,
  onEdit,
  onDelete,
  onMention,
  currentUser,
}) => {
  const [isEditing, setIsEditing] = React.useState(false);
  const [editContent, setEditContent] = React.useState(content);
  const [showDeleteConfirm, setShowDeleteConfirm] = React.useState(false);
  const [loading, setLoading] = React.useState<'edit' | 'delete' | null>(null);

  const canModify = currentUser?.id === author.id;
  const showActions = canModify && (canEdit || canDelete);

  const formatTime = (date: Date) => {
    return new Intl.DateTimeFormat('zh-CN', {
      month: 'short',
      day: 'numeric',
      hour: '2-digit',
      minute: '2-digit',
    }).format(date);
  };

  const handleEdit = async () => {
    if (editContent.trim() === content) {
      setIsEditing(false);
      return;
    }
    setLoading('edit');
    try {
      await onEdit(id, editContent.trim());
      setIsEditing(false);
    } finally {
      setLoading(null);
    }
  };

  const handleDelete = async () => {
    setLoading('delete');
    try {
      await onDelete(id);
    } finally {
      setLoading(null);
      setShowDeleteConfirm(false);
    }
  };

  const handleCancel = () => {
    setIsEditing(false);
    setEditContent(content);
  };

  // 解析 @mentions
  const renderContent = (text: string) => {
    const mentionRegex = /@(\[[^\]]+\]\(([^)]+)\))/g;
    const parts = text.split(mentionRegex);

    return parts.map((part, index) => {
      if (index % 3 === 2 && onMention) {
        return (
          <button
            key={index}
            className="comment-item__mention"
            onClick={() => onMention(part)}
            type="button"
          >
            @{part}
          </button>
        );
      }
      if (index % 3 === 1) return null;
      return <span key={index}>{part}</span>;
    });
  };

  return (
    <div
      className={`comment-item ${isEditing ? 'comment-item--editing' : ''}`}
      role="article"
      aria-label={`${author.name}的评论`}
    >
      <Avatar user={author} size="md" aria-hidden="true" />

      <div className="comment-item__body">
        <div className="comment-item__header">
          <span className="comment-item__author">{author.name}</span>
          <time
            className="comment-item__time"
            dateTime={createdAt.toISOString()}
          >
            {formatTime(createdAt)}
          </time>
          {editedAt && (
            <span className="comment-item__edited" title={`编辑于${formatTime(editedAt)}`}>
              (已编辑)
            </span>
          )}
        </div>

        {/* 显示模式 */}
        {!isEditing && (
          <div className="comment-item__content">
            {renderContent(content)}
          </div>
        )}

        {/* 编辑模式 */}
        {isEditing && (
          <div className="comment-item__editor">
            <TextArea
              value={editContent}
              onChange={setEditContent}
              rows={3}
              autoFocus
              aria-label="编辑评论"
            />
            <div className="comment-item__editor-actions">
              <Button
                size="sm"
                onClick={handleEdit}
                loading={loading === 'edit'}
              >
                保存
              </Button>
              <Button
                size="sm"
                variant="ghost"
                onClick={handleCancel}
                disabled={loading === 'edit'}
              >
                取消
              </Button>
            </div>
          </div>
        )}

        {/* 操作按钮 */}
        {showActions && !isEditing && (
          <div className="comment-item__actions">
            {canEdit && (
              <button
                className="comment-item__action"
                onClick={() => setIsEditing(true)}
                disabled={loading !== null}
              >
                编辑
              </button>
            )}
            {canDelete && (
              <>
                {!showDeleteConfirm ? (
                  <button
                    className="comment-item__action comment-item__action--danger"
                    onClick={() => setShowDeleteConfirm(true)}
                    disabled={loading !== null}
                  >
                    删除
                  </button>
                ) : (
                  <div className="comment-item__delete-confirm">
                    <span>确认删除？</span>
                    <button
                      className="comment-item__confirm"
                      onClick={handleDelete}
                      loading={loading === 'delete'}
                    >
                      确认
                    </button>
                    <button
                      className="comment-item__cancel"
                      onClick={() => setShowDeleteConfirm(false)}
                      disabled={loading === 'delete'}
                    >
                      取消
                    </button>
                  </div>
                )}
              </>
            )}
          </div>
        )}
      </div>
    </div>
  );
};
```
