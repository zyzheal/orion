## 问题描述

**原始问题**: 效能看板等图表无 alt 文本，视障用户无法理解数据

**影响页面**:
- 效能看板 (DORA 指标趋势图)
- 流水线运行统计图
- 审批效率分析图
- 成本趋势图

## 解决方案

### 1. 创建 ChartAccessibility 组件

```tsx
// orion-dba/frontend/src/components/Chart/ChartAccessibility.tsx
import React, { useState } from 'react';
import { ChartData } from './types';

interface ChartAccessibilityProps {
  chartTitle: string;
  chartType: 'line' | 'bar' | 'pie' | 'area';
  data: ChartData[];
  summary?: string;
  trend?: 'up' | 'down' | 'stable';
}

/**
 * 图表无障碍访问组件
 * 提供屏幕阅读器友好的文本描述和数据表格
 */
export const ChartAccessibility: React.FC<ChartAccessibilityProps> = ({
  chartTitle,
  chartType,
  data,
  summary,
  trend,
}) => {
  const [showTable, setShowTable] = useState(false);

  // 生成图表长描述
  const generateLongDescription = () => {
    const trendText = trend ? 
      (trend === 'up' ? '呈上升趋势' : trend === 'down' ? '呈下降趋势' : '保持稳定') : 
      '';
    
    const dataPoints = data.slice(0, 5).map(d => `${d.label}: ${d.value}`).join(', ');
    const moreText = data.length > 5 ? `等${data.length}个数据点` : '';
    
    return `${chartTitle}是一个${chartType}图。${trendText}数据显示：${dataPoints}${moreText}。${summary || ''}`;
  };

  // 计算统计信息
  const stats = {
    total: data.reduce((sum, d) => sum + d.value, 0),
    average: data.reduce((sum, d) => sum + d.value, 0) / data.length,
    max: Math.max(...data.map(d => d.value)),
    min: Math.min(...data.map(d => d.value)),
  };

  // 生成数据表格
  const renderDataTable = () => (
    <table className="chart-data-table" aria-label={`${chartTitle} 数据明细`}>
      <caption>{chartTitle} - 详细数据</caption>
      <thead>
        <tr>
          <th scope="col">类别</th>
          <th scope="col">数值</th>
          <th scope="col">占比</th>
          <th scope="col">趋势</th>
        </tr>
      </thead>
      <tbody>
        {data.map((d, i) => (
          <tr key={i}>
            <td>{d.label}</td>
            <td>{d.value}</td>
            <td>{d.percentage?.toFixed(1) || '-'}%</td>
            <td>
              {d.trend === 'up' && '↑ 增长'}
              {d.trend === 'down' && '↓ 下降'}
              {d.trend === 'stable' && '→ 持平'}
            </td>
          </tr>
        ))}
      </tbody>
      <tfoot>
        <tr>
          <th scope="row">总计</th>
          <td>{stats.total}</td>
          <td>100%</td>
          <td>-</td>
        </tr>
        <tr>
          <th scope="row">平均值</th>
          <td>{stats.average.toFixed(2)}</td>
          <td>-</td>
          <td>-</td>
        </tr>
      </tfoot>
    </table>
  );

  return (
    <div className="chart-accessibility" role="region" aria-label={`${chartTitle} 无障碍访问`}>
      {/* 短描述 (aria-label) */}
      <span className="sr-only" role="text" aria-label={chartTitle}>
        {chartTitle}
      </span>
      
      {/* 长描述 (aria-describedby) */}
      <div 
        id={`chart-desc-${chartTitle.replace(/\s+/g, '-').toLowerCase()}`} 
        className="sr-only"
      >
        {generateLongDescription()}
      </div>
      
      {/* 统计摘要 */}
      <div className="chart-stats-summary" aria-hidden="true">
        <h4 className="sr-only">统计摘要</h4>
        <dl className="stats-list">
          <div className="stat-item">
            <dt>总数</dt>
            <dd>{stats.total}</dd>
          </div>
          <div className="stat-item">
            <dt>平均值</dt>
            <dd>{stats.average.toFixed(2)}</dd>
          </div>
          <div className="stat-item">
            <dt>最大值</dt>
            <dd>{stats.max}</dd>
          </div>
          <div className="stat-item">
            <dt>最小值</dt>
            <dd>{stats.min}</dd>
          </div>
        </dl>
      </div>
      
      {/* 数据表格切换 */}
      <div className="chart-data-table-container">
        <button 
          onClick={() => setShowTable(!showTable)}
          aria-expanded={showTable}
          aria-controls={`data-table-${chartTitle.replace(/\s+/g, '-').toLowerCase()}`}
          className="btn-toggle-table"
        >
          {showTable ? '隐藏数据表格' : '查看数据表格'}
        </button>
        
        {showTable && (
          <div id={`data-table-${chartTitle.replace(/\s+/g, '-').toLowerCase()}`}>
            {renderDataTable()}
          </div>
        )}
      </div>
      
      {/* 下载数据 */}
      <div className="chart-download-options">
        <a 
          href={`/api/charts/${chartTitle}/data.csv`}
          download={`${chartTitle}-data.csv`}
          className="btn-download-csv"
          aria-label={`下载${chartTitle}CSV 格式数据`}
        >
          📥 下载 CSV
        </a>
        <a 
          href={`/api/charts/${chartTitle}/data.xlsx`}
          download={`${chartTitle}-data.xlsx`}
          className="btn-download-xlsx"
          aria-label={`下载${chartTitle}Excel 格式数据`}
        >
          📊 下载 Excel
        </a>
      </div>
    </div>
  );
};

export default ChartAccessibility;
```

### 2. 集成到图表组件

```tsx
// orion-dba/frontend/src/components/Chart/AccessibleLineChart.tsx
import React from 'react';
import { LineChart, Line, XAxis, YAxis, Tooltip, ResponsiveContainer } from 'recharts';
import { ChartAccessibility } from './ChartAccessibility';
import { ChartData } from './types';

interface AccessibleLineChartProps {
  title: string;
  data: ChartData[];
  summary?: string;
  trend?: 'up' | 'down' | 'stable';
}

export const AccessibleLineChart: React.FC<AccessibleLineChartProps> = ({
  title,
  data,
  summary,
  trend,
}) => {
  const chartId = title.replace(/\s+/g, '-').toLowerCase();

  return (
    <div 
      className="chart-container"
      role="img"
      aria-label={title}
      aria-describedby={`chart-desc-${chartId}`}
    >
      {/* 图表标题 */}
      <h3 className="chart-title" id={`chart-title-${chartId}`}>
        {title}
      </h3>
      
      {/* 图表渲染 */}
      <ResponsiveContainer width="100%" height={300}>
        <LineChart data={data} aria-label={`${title}折线图`}>
          <XAxis 
            dataKey="label" 
            aria-label="X 轴：时间"
          />
          <YAxis 
            aria-label="Y 轴：数值"
          />
          <Tooltip 
            content={({ active, payload }) => {
              if (active && payload && payload.length) {
                return (
                  <div className="tooltip" role="tooltip">
                    <p>{payload[0].payload.label}</p>
                    <p>数值：{payload[0].value}</p>
                  </div>
                );
              }
              return null;
            }}
          />
          <Line 
            type="monotone" 
            dataKey="value" 
            stroke="#0070F3"
            aria-label="数据趋势线"
          />
        </LineChart>
      </ResponsiveContainer>
      
      {/* 无障碍访问组件 */}
      <ChartAccessibility
        chartTitle={title}
        chartType="line"
        data={data}
        summary={summary}
        trend={trend}
      />
    </div>
  );
};
```

### 3. 应用到效能看板

```tsx
// orion-dba/frontend/src/views/dashboard/EfficiencyDashboard.tsx
import React from 'react';
import { AccessibleLineChart } from '@/components/Chart/AccessibleLineChart';
import { AccessibleBarChart } from '@/components/Chart/AccessibleBarChart';
import { useDoraMetrics } from '@/hooks/useDoraMetrics';

export const EfficiencyDashboard: React.FC = () => {
  const { deploymentFrequency, leadTime, mttr, changeFailureRate } = useDoraMetrics();

  return (
    <div className="efficiency-dashboard" role="main" aria-label="效能看板">
      <h1>效能看板</h1>
      
      {/* 部署频率趋势 */}
      <AccessibleLineChart
        title="部署频率趋势"
        data={deploymentFrequency}
        summary="显示最近 4 周的部署频率，从每周 2.1 次增长到 3.5 次"
        trend="up"
      />
      
      {/* 前置时间趋势 */}
      <AccessibleLineChart
        title="代码前置时间"
        data={leadTime}
        summary="显示代码提交到部署的平均时间，从 2.5 小时降低到 1.8 小时"
        trend="down"
      />
      
      {/* 恢复时间趋势 */}
      <AccessibleLineChart
        title="平均恢复时间 (MTTR)"
        data={mttr}
        summary="显示故障发生到恢复的平均时间，保持在 30 分钟左右"
        trend="stable"
      />
      
      {/* 变更失败率 */}
      <AccessibleBarChart
        title="变更失败率"
        data={changeFailureRate}
        summary="显示部署导致故障的比例，从 15% 降低到 8%"
        trend="down"
      />
    </div>
  );
};
```
