## 问题描述

**原始问题**: WebSocket 推送、流水线状态更新等动态内容变化时，屏幕阅读器用户无法感知

**影响场景**:
- 流水线状态实时更新
- 审批请求到达
- WebSocket 消息推送
- 告警通知
- 表单提交结果反馈

## 解决方案

### 1. 创建 ScreenReaderAnnouncer 组件

```tsx
// orion-dba/frontend/src/components/ScreenReaderAnnouncer.tsx
import React, { useEffect, useRef } from 'react';

interface ScreenReaderAnnouncerProps {
  message: string;
  priority?: 'polite' | 'assertive';
  onAnnounce?: () => void;
}

/**
 * 屏幕阅读器通知组件
 * 使用 aria-live 区域动态通知屏幕阅读器用户
 */
export const ScreenReaderAnnouncer: React.FC<ScreenReaderAnnouncerProps> = ({
  message,
  priority = 'polite',
  onAnnounce,
}) => {
  const announcerRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (announcerRef.current && message) {
      // 设置通知优先级
      announcerRef.current.setAttribute('aria-live', priority);
      
      // 通知屏幕阅读器
      announcerRef.current.textContent = message;
      
      // 回调
      if (onAnnounce) {
        setTimeout(onAnnounce, 1000);
      }
    }
  }, [message, priority, onAnnounce]);

  return (
    <div
      ref={announcerRef}
      role="status"
      aria-live={priority}
      aria-atomic="true"
      style={{
        position: 'absolute',
        width: '1px',
        height: '1px',
        padding: '0',
        margin: '-1px',
        overflow: 'hidden',
        clip: 'rect(0, 0, 0, 0)',
        whiteSpace: 'nowrap',
        border: '0',
      }}
    />
  );
};

export default ScreenReaderAnnouncer;
```

### 2. 封装工具函数

```typescript
// orion-dba/frontend/src/utils/accessibility.ts

/**
 * 通知屏幕阅读器
 * @param message 通知内容
 * @param priority 优先级 ('polite' | 'assertive')
 */
export function notifyScreenReader(
  message: string,
  priority: 'polite' | 'assertive' = 'polite'
): void {
  const announcer = document.getElementById('sr-announcer');
  if (announcer) {
    announcer.setAttribute('aria-live', priority);
    announcer.textContent = message;
  }
}

/**
 * 通知屏幕阅读器 - 简短消息
 */
export function notifyBrief(message: string): void {
  notifyScreenReader(message, 'polite');
}

/**
 * 通知屏幕阅读器 - 重要消息
 */
export function notifyImportant(message: string): void {
  notifyScreenReader(message, 'assertive');
}

/**
 * 通知屏幕阅读器 - 错误消息
 */
export function notifyError(message: string): void {
  notifyScreenReader(`错误：${message}`, 'assertive');
}

/**
 * 通知屏幕阅读器 - 成功消息
 */
export function notifySuccess(message: string): void {
  notifyScreenReader(`成功：${message}`, 'polite');
}
```

```tsx

/**

export const GlobalAnnouncer: React.FC = () => {

  return (

      role="status"

        overflow: 'hidden',
        clip: 'rect(0, 0, 0, 0)',

        border: '0',
      }}
    >
      {message}
    </div>
  );
};
```

### 4. 状态管理

```typescript
// orion-dba/frontend/src/store/announcer.ts
import { create } from 'zustand';

interface AnnouncerState {
  message: string;
  priority: 'polite' | 'assertive';
  announce: (message: string, priority?: 'polite' | 'assertive') => void;
}

export const useAnnouncerStore = create<AnnouncerState>((set) => ({
  message: '',
  priority: 'polite',
  announce: (message, priority = 'polite') => {
    set({ message, priority });
    // 1 秒后清空，允许重复通知
    setTimeout(() => set({ message: '' }), 1000);
  },
}));
```

### 3. WebSocket 消息推送

```tsx
// orion-dba/frontend/src/hooks/useWebSocket.ts
import { notifyScreenReader } from '@/utils/accessibility';

export function useWebSocket(channel: string) {
  useEffect(() => {
    const ws = new WebSocket(`ws://${API_BASE}/${channel}`);
    
    ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      
      // 处理消息
      handleMessage(data);
      
      // 通知屏幕阅读器
      if (data.type === 'notification') {
        notifyScreenReader(data.content, 'polite');
      } else if (data.type === 'alert') {
        notifyImportant(data.content);
      }
    };

    return () => ws.close();
  }, [channel]);
}
```

### 4. 表单提交反馈

// orion-dba/frontend/src/components/Form/FormSubmit.tsx
import { notifySuccess, notifyError } from '@/utils/accessibility';

function FormSubmit({ onSubmit, loading, children }) {
  const handleSubmit = async (values) => {
    try {
      await onSubmit(values);
      notifySuccess('提交成功');
    } catch (error) {
      notifyError(`提交失败：${error.message}`);
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      {children}
    </form>
  );
}
