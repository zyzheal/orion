### 3.3 ROI 动态计算模型

```python
class ROICalculator:
    """
    ROI 动态计算模型
    """
    
    def calculate_monthly_roi(self, team_count: int, pr_count: int) -> dict:
        # 成本计算
        infrastructure_cost = self._calculate_infrastructure_cost(team_count)
        ai_cost = self._calculate_ai_cost(pr_count)
        labor_cost = self._calculate_labor_cost()
        total_cost = infrastructure_cost + ai_cost + labor_cost
        
        # 收益计算
        ci_saving = self._calculate_ci_saving(pr_count)
        incident_reduction = self._calculate_incident_reduction()
        sre_efficiency = self._calculate_sre_efficiency()
        resource_optimization = self._calculate_resource_optimization()
        total_benefit = ci_saving + incident_reduction + sre_efficiency + resource_optimization
        
        # ROI 计算
        roi = (total_benefit - total_cost) / total_cost * 100
        payback_months = total_cost / (total_benefit / 30)  # 回本周期 (月)
        
        return {
            "cost": {
                "infrastructure": infrastructure_cost,
                "ai": ai_cost,
                "labor": labor_cost,
                "total": total_cost,
            },
            "benefit": {
                "ci_saving": ci_saving,
                "incident_reduction": incident_reduction,
                "sre_efficiency": sre_efficiency,
                "resource_optimization": resource_optimization,
                "total": total_benefit,
            },
            "roi": round(roi, 2),
            "payback_months": round(payback_months, 1),
        }
    
    def _calculate_infrastructure_cost(self, team_count: int) -> float:
        # 基础设施成本: 每团队 ~400 元/月
        return team_count * 400
    
    def _calculate_ai_cost(self, pr_count: int) -> float:
        # AI 成本: 每 PR ~5 元 (LLM + 向量 + 图)
        return pr_count * 5
    
    def _calculate_ci_saving(self, pr_count: int) -> float:
        # CI 时间节省: 每 PR 节省 27 分钟 × 人力成本
        # 假设工程师时薪 100 元，节省 27 分钟 = 45 元/PR
        return pr_count * 45
```

### 3.4 租户成本分摊算法

```python
class TenantCostAllocator:
    """
    多租户成本分摊算法
    """
    
    def allocate_costs(self, tenants: list[Tenant], total_costs: dict) -> dict:
        """
        按资源使用比例分摊成本
        """
        # 计算总资源使用
        total_cpu = sum(t.resource_usage.cpu for t in tenants)
        total_memory = sum(t.resource_usage.memory for t in tenants)
        total_storage = sum(t.resource_usage.storage for t in tenants)
        
        # 按权重分摊
        allocated_costs = {}
        for tenant in tenants:
            cpu_ratio = tenant.resource_usage.cpu / total_cpu
            memory_ratio = tenant.resource_usage.memory / total_memory
            storage_ratio = tenant.resource_usage.storage / total_storage
            
            allocated_costs[tenant.id] = {
                "compute": (
                    cpu_ratio * total_costs["compute"] * 0.5 +
                    memory_ratio * total_costs["compute"] * 0.5
                ),
                "storage": storage_ratio * total_costs["storage"],
                "ai": tenant.ai_usage.token_count * self.ai_token_price,
                "network": tenant.network_usage.bandwidth * self.network_price,
            }
        
        return allocated_costs
```

   ├── 变更摘要
   ├── 风险评分 (0-100)
   ├── 风险因子解释
   ├── 成本影响分析
   ├── 建议 (优化/修复)
   └── 审批要求 (根据风险等级)

5. 审批流程
   ├── 低风险 (0-30): 自动通过

### 4.2 成本预估模型

```python
class IaCCostEstimator:
    """
    IaC 成本预估模型
    """
    
    def estimate_cost(self, plan: TerraformPlan) -> CostEstimate:
        """
        预估 Terraform 变更的成本影响
        """
        current_cost = self._get_current_cost()
        estimated_cost = self._calculate_estimated_cost(plan)
        
        return CostEstimate(
            current_monthly=current_cost,
            estimated_monthly=estimated_cost,
            delta=estimated_cost - current_cost,
            delta_percent=(estimated_cost - current_cost) / current_cost * 100,
            breakdown=self._get_cost_breakdown(plan),
        )
    
    def _calculate_estimated_cost(self, plan: TerraformPlan) -> float:
        total = 0
        for resource in plan.resource_changes:
            if resource.action == "create":
                total += self._get_resource_monthly_cost(resource)
            elif resource.action == "update":
                old_cost = self._get_resource_monthly_cost(resource.before)
                new_cost = self._get_resource_monthly_cost(resource.after)
                total += new_cost - old_cost
            elif resource.action == "delete":
                total -= self._get_resource_monthly_cost(resource.before)
        return total
```
