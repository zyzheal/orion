interface SystemConfig {
  // 核心配置
  core: {
    app: { port, host, env, logLevel };

  pipeline: {
    maxConcurrentRuns: number;

  artifact: { maxSize, retentionDays };

    providers: { openai: {}, anthropic: {} };
    limits: { rateLimit, maxTokens };
    embedding: { dimension, model };
  };
  aiReview: { timeout, maxFileSize };
  aiSecurity: { blockedPatterns, maxInputLength };
  llmTrace: { retentionDays, maxSpans };
  modelVersion: { maxVersions, modelTypes };

  // 运维域

  approval: { timeout, maxLevels };
  qualityGate: { thresholds };

  // 安全域
  security: { policies };
  audit: { retentionDays, hashAlgorithm };
  tenant: { maxTenants, quota };
  privacy: { maskingRules };
  
  // 平台域
  cmdb: { syncInterval };
  plugin: { maxSize, maxCount };
  webhook: { timeout };
  queue: { configs };
  session: { ttl };
  user: { maxPerTenant };

  digitalTwin: { configs };
  federation: { configs };
  multiCloud: { configs };

## 五、实施建议

### 5.1 分阶段迁移

| 阶段 | 模块 | 优先级 |
|------|------|--------|
| Phase 1 | 核心域 (pipeline, build, deploy) | P0 |
| Phase 2 | 运维域 (alert, monitoring, self-healing) | P0 |
| Phase 3 | 业务域 (ticketing, chatops, notification) | P1 |
| Phase 4 | 安全域 (security, audit, tenant) | P1 |
| Phase 5 | 平台域 + 扩展域 | P2 |

### 5.2 统一配置覆盖

所有 92 个模块的硬编码配置，都应迁移到 `UnifiedConfigService`，实现：
- 统一存储 (PostgreSQL)
- 统一 API
- 统一变更历史
- 统一热更新
