### 1.2 EventSource (SSE) 设计

```typescript
// EventSource 用于单向推送 (适合日志流)

class LogStream {
  private eventSource: EventSource;
  
  constructor(runId: string, stageName: string) {
    const url = `/api/v1/pipelines/${runId}/stages/${stageName}/logs/stream`;
    this.eventSource = new EventSource(url);
  }
  
  onLog(handler: (log: LogEntry) => void) {
    this.eventSource.addEventListener("log", (event) => {
      handler(JSON.parse(event.data));
    });
  }
  
  onComplete(handler: () => void) {
    this.eventSource.addEventListener("complete", handler);
  }
  
  onError(handler: (error: Error) => void) {
    this.eventSource.addEventListener("error", handler);
  }
  
  close() {
    this.eventSource.close();
  }
}

// 使用示例
const logStream = new LogStream("run-478", "scan");
logStream.onLog((log) => {
  appendToLog(log.line);
});
logStream.onComplete(() => {
  logStream.close();
  markStageComplete();
});
```

```python
# FastAPI WebSocket 实现

from fastapi import WebSocket, WebSocketDisconnect
from typing import Dict, Set

class ConnectionManager:
    def __init__(self):
        self.active_connections: Dict[str, Set[WebSocket]] = {}
    
    async def connect(self, websocket: WebSocket, channel: str):
        await websocket.accept()
        if channel not in self.active_connections:
            self.active_connections[channel] = set()
        self.active_connections[channel].add(websocket)
    
    def disconnect(self, websocket: WebSocket, channel: str):
        if channel in self.active_connections:
            self.active_connections[channel].discard(websocket)
    
    async def broadcast(self, channel: str, message: dict):
        if channel in self.active_connections:
            disconnected = set()
            for connection in self.active_connections[channel]:
                try:
                    await connection.send_json(message)
                except:
                    disconnected.add(connection)
            # 清理断开的连接
            for conn in disconnected:
                self.active_connections[channel].discard(conn)

manager = ConnectionManager()

@app.websocket("/ws")
async def websocket_endpoint(websocket: WebSocket, channel: str):
    await manager.connect(websocket, channel)
    try:
        while True:
            # 保持连接，接收心跳
            data = await websocket.receive_text()
            if data == "ping":
                await websocket.send_text("pong")
    except WebSocketDisconnect:
        manager.disconnect(websocket, channel)

# 发布事件到 WebSocket
async def publish_event(channel: str, event_type: str, data: dict):
    await manager.broadcast(channel, {
        "type": event_type,
        "data": data,
        "timestamp": datetime.utcnow().isoformat(),
    })
```

### 2.2 通知聚合策略

```python
class NotificationAggregator:
    """
    通知聚合策略
    """
    
    def aggregate(self, notifications: list[Notification]) -> list[Notification]:
        """
        聚合相似通知
        """
        # 按类型分组
        grouped = self._group_by_type(notifications)
        
        aggregated = []
        for group_type, group_notifications in grouped.items():
            if len(group_notifications) == 1:
                aggregated.extend(group_notifications)
            else:
                # 聚合为一条
                aggregated.append(self._create_aggregated_notification(
                    group_type,
                    group_notifications
                ))
        
        return aggregated
    
    def _group_by_type(self, notifications: list[Notification]) -> dict:
        groups = {}
        for n in notifications:
            key = f"{n.type}:{n.resource_id}"
            if key not in groups:
                groups[key] = []
            groups[key].append(n)
        return groups
    
    def _create_aggregated_notification(self, group_type: str, 
                                        notifications: list[Notification]) -> Notification:
        return Notification(
            type="aggregated",
            title=f"{len(notifications)} 条{group_type}通知",
            summary=f"从 {notifications[0].timestamp} 到 {notifications[-1].timestamp}",
            count=len(notifications),
            notifications=notifications,
        )
```

```python
    async def check_timeout(self, approval: Approval):
        """
        检查审批是否超时
        """
        risk_level = approval.risk_level
        rule = self.TIMEOUT_RULES[risk_level]
        
        elapsed = datetime.utcnow() - approval.created_at
        if elapsed > timedelta(hours=rule["hours"]):
            if rule["action"] == "auto_approve":
                await self._auto_approve(approval)
            elif rule["action"] == "escalate":
                await self._escalate(approval)
    
    async def _escalate(self, approval: Approval):
        """
        升级到上级
        """
        # 通知上级
        manager = await self._get_manager(approval.approver)
        await self._notify(manager, f"审批 {approval.id} 已超时，请处理")
        
        # 如果再次超时，自动拒绝
        approval.escalated = True
        approval.escalated_at = datetime.utcnow()
```

```typescript
// 状态标识组件

type StatusType = 'success' | 'warning' | 'error' | 'info' | 'pending';

interface StatusBadgeProps {
  status: StatusType;
  label: string;
  icon?: boolean;
  size?: 'small' | 'medium' | 'large';
}

const StatusBadge: React.FC<StatusBadgeProps> = ({ 
  status, 
  label, 
  icon = true,
  size = 'medium' 
}) => {
  const config = {
    success: { color: '#52c41a', icon: '✅', bg: '#f6ffed' },
    warning: { color: '#faad14', icon: '⚠️', bg: '#fffbe6' },
    error: { color: '#ff4d4f', icon: '❌', bg: '#fff2f0' },
    info: { color: '#1890ff', icon: 'ℹ️', bg: '#e6f7ff' },
    pending: { color: '#faad14', icon: '⏳', bg: '#fffbe6' },
  };
  
  const { color, icon: statusIcon, bg } = config[status];
  
  return (
    <span 
      style={{ 
        color, 
        backgroundColor: bg,
        padding: size === 'small' ? '2px 8px' : '4px 12px',
        borderRadius: '4px',
        fontSize: size === 'small' ? '12px' : '14px',
      }}
    >
      {icon && <span style={{ marginRight: '4px' }}>{statusIcon}</span>}
      {label}
    </span>
  );
};
```
