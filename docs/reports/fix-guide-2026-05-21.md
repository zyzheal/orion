## 🔴 P0-1: 缺少 Loading 状态 (110 个)

### 问题描述
异步操作函数没有对应的 loading 状态，导致用户可以重复点击提交按钮。

### 修复模板

```typescript
// ❌ 修复前
const handleSubmit = async () => {
  await api.save(data);
  message.success('保存成功');
};

// ✅ 修复后
const [loading, setLoading] = useState(false);

const handleSubmit = async () => {
  setLoading(true);
  try {
    await api.save(data);
    message.success('保存成功');
  } catch (error) {
    message.error('保存失败');
  } finally {
    setLoading(false);
  }
};

// 按钮使用
<Button loading={loading} onClick={handleSubmit}>
  提交
</Button>
```

### 快速修复命令
```bash
# 使用 AST 分析器自动修复
npx tsx docs/design-constraints/framework/core/auto-fix.ts --type loading
```

## 🔴 P0-2: 缺少操作反馈 (15 个)

### 问题描述
异步操作没有 success/error 提示，用户不知道操作是否成功。

### 修复模板

```typescript
// ❌ 修复前
const handleDelete = async (id) => {
  await api.delete(id);
};

// ✅ 修复后
const handleDelete = async (id) => {
  try {
    await api.delete(id);
    message.success('删除成功');
    refreshList();
  } catch (error) {
    message.error('删除失败: ' + error.message);
  }
};
```

### 推荐反馈写法
```typescript
// 成功
message.success('操作成功');

// 失败
message.error('操作失败: ' + error.message);

// 带描述
message.success({
  content: '保存成功',
  description: '数据已更新',
});
```

## 🔴 P0-3: as any 类型断言 (80+ 处)

### 问题描述
使用 `as any` 绕过 TypeScript 类型检查，降低代码类型安全性。

### 修复模板

```typescript
// ❌ 修复前
const data = response.data as any;

// ✅ 修复后 - 定义接口
interface ApiResponse<T> {
  code: number;
  data: T;
  message?: string;
}

interface User {
  id: string;
  name: string;
}

// 使用
const data = (response.data as ApiResponse<User>).data;

// ❌ 修复前 2
const result = data as any;

// ✅ 修复后 2 - 类型守卫
function isUser(data: unknown): data is User {
  return typeof data === 'object'
    && data !== null
    && 'id' in data
    && 'name' in data;
}

if (isUser(data)) {
  console.log(data.name); // TypeScript 已知是 User
}
```

### 推荐类型库

```typescript
// types/api.ts
export interface ApiResponse<T> {
  code: number;
  data: T;
  message?: string;
  timestamp?: string;
}

export interface PaginatedResponse<T> {
  data: T[];
  total: number;
  page: number;
  pageSize: number;
}

export interface ErrorResponse {
  code: number;
  message: string;
  details?: Record<string, string>;
}
```

## 🟡 P1-1: 缺少提交按钮 (22 个)

### 问题描述
Form 组件没有提交按钮，用户无法提交表单。

### 修复模板

```typescript
// ❌ 修复前
<Form layout="vertical">
  <Form.Item label="名称" name="name">
    <Input />
  </Form.Item>
  {/* 缺少提交按钮 */}
</Form>

// ✅ 修复后
<Form layout="vertical" form={form} onFinish={handleSubmit}>
  <Form.Item label="名称" name="name" rules={[{ required: true }]}>
    <Input />
  </Form.Item>

  <Form.Item>
    <Space>
      <Button type="primary" htmlType="submit">
        提交
      </Button>
      <Button onClick={onCancel}>取消</Button>
    </Space>
  </Form.Item>
</Form>
```

### Modal 表单

```typescript
<Modal
  open={open}
  onCancel={onClose}
  footer={[
    <Button key="cancel" onClick={onClose}>取消</Button>,
    <Button key="submit" type="primary" htmlType="submit">确定</Button>,
  ]}
>
  <Form layout="vertical" onFinish={handleSubmit}>
    {/* 表单内容 */}
  </Form>
</Modal>
```

## 🟡 P1-2: 缺少编辑入口 (5 个)

### 问题描述
详情页只有查看模式，没有编辑入口。

### 修复模板

```typescript
// ❌ 修复前
const DetailPage = () => (
  <Descriptions>
    <Descriptions.Item label="名称">{data.name}</Descriptions.Item>
  </Descriptions>
);

// ✅ 修复后
const DetailPage = () => {
  const [editing, setEditing] = useState(false);

  return (
    <div>
      <Space style={{ marginBottom: 16 }}>
        <Button
          type="primary"
          icon={<EditOutlined />}
          onClick={() => setEditing(!editing)}
        >
          {editing ? '取消编辑' : '编辑'}
        </Button>
      </Space>

      {editing ? (
        <Form form={form} layout="vertical" initialValues={data}>
          <Form.Item label="名称" name="name">
            <Input />
          </Form.Item>
          <Form.Item>
            <Space>
              <Button type="primary" onClick={handleSave}>保存</Button>
              <Button onClick={() => setEditing(false)}>取消</Button>
            </Space>
          </Form.Item>
        </Form>
      ) : (
        <Descriptions>
          <Descriptions.Item label="名称">{data.name}</Descriptions.Item>
        </Descriptions>
      )}
    </div>
  );
};
```

## 🟡 P1-3: 硬编码颜色 (28 处)

### 问题描述
使用硬编码颜色值而非 Design Token，不利于主题切换和维护。

### 修复模板

```typescript
// ❌ 修复前
<div style={{ color: '#3370E6' }}>主色</div>
<div style={{ color: '#52c41a' }}>成功</div>
<div style={{ color: '#f5222d' }}>错误</div>

// ✅ 修复后
import { colors, spacing } from '@/tokens';

<div style={{ color: colors.primary[500] }}>主色</div>
<div style={{ color: colors.success[500] }}>成功</div>
<div style={{ color: colors.error[500] }}>错误</div>
```

### Design Token 速查表

| 用途 | Token | 值 |
|------|-------|-----|
| 主色 | `colors.primary[500]` | #3370E6 |
| 成功 | `colors.success[500]` | #52c41a |
| 警告 | `colors.warning[500]` | #faad14 |
| 错误 | `colors.error[500]` | #f5222d |
| 信息 | `colors.info[500]` | #3a98f4 |
| 中性灰 | `colors.neutral[500]` | #8c8c8c |
| 间距-小 | `spacing.sm` | 8px |
| 间距-中 | `spacing.md` | 16px |
| 间距-大 | `spacing.lg` | 24px |
| 圆角-按钮 | `componentRadius.button` | 6px |
| 圆角-卡片 | `componentRadius.card` | 12px |
