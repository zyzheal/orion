# 周报导出与分享设计

> 版本：v1.0 | 创建日期：2026-04-11 | 状态：草案

---

## 1. 概述

### 1.1 设计目标

提供周报的多格式导出和分享功能，支持 PDF、Excel、邮件等多种方式，满足不同场景需求。

**对应需求**: FR-1.8 — 作为 Tech Lead，我想要自动周报，以便节省统计时间（含导出功能）

### 1.2 导出格式

| 格式 | 用途 | 优先级 |
|------|------|--------|
| **PDF** | 正式报告、存档、打印 | P0 |
| **Excel** | 数据分析、二次加工 | P0 |
| **HTML** | 邮件正文、网页查看 | P0 |
| **Markdown** | 知识库归档、Git 提交 | P1 |
| **PNG** | 社交媒体分享、快速预览 | P1 |

### 1.3 分享渠道

| 渠道 | 方式 | 优先级 |
|------|------|--------|
| **邮件** | 附件 + 正文 | P0 |
| **钉钉** | 文件 + 卡片 | P0 |
| **企微** | 文件 + 卡片 | P0 |
| **链接分享** | 加密链接 | P1 |
| **知识库** | 自动归档 | P1 |

### 2.2 PDF 生成技术

```yaml
PDF 生成技术栈:
  方案选择: wkhtmltopdf (HTML to PDF)
  
  理由:
    - 支持 CSS 样式，保持与网页一致
    - 支持图表渲染 (Chart.js/ECharts)
    - 支持页眉页脚
    - 支持目录自动生成
    - 开源免费
  
  流程:
    1. 渲染 HTML 模板 (Go template)
    2. 加载 CSS 样式
    3. 渲染图表 (等待 JS 执行)
    4. 调用 wkhtmltopdf 生成 PDF
    5. 优化 PDF (压缩、添加书签)
  
  配置:
    page_size: A4
    margin_top: 20mm
    margin_bottom: 20mm
    margin_left: 15mm
    margin_right: 15mm
    encoding: UTF-8
    enable_javascript: true
    javascript_delay: 1000  # 等待图表渲染
```

### 2.3 PDF 生成代码

```go
package report

import (
    "os/exec"
    "path/filepath"
)

type PDFGenerator struct {
    templateDir string
    outputDir   string
}

func (g *PDFGenerator) GenerateWeeklyReport(report ReportData) (string, error) {
    // 1. 渲染 HTML
    htmlPath, err := g.renderHTML(report)
    if err != nil {
        return "", err
    }
    
    // 2. 生成 PDF
    pdfPath := filepath.Join(g.outputDir, report.WeekID+".pdf")
    
    cmd := exec.Command("wkhtmltopdf",
        "--page-size", "A4",
        "--margin-top", "20mm",
        "--margin-bottom", "20mm",
        "--margin-left", "15mm",
        "--margin-right", "15mm",
        "--encoding", "UTF-8",
        "--enable-javascript",
        "--javascript-delay", "1000",
        "--header-center", "[title]",
        "--header-font-size", "10",
        "--footer-center", "[page] / [toPage]",
        "--footer-font-size", "8",
        htmlPath,
        pdfPath,
    )
    
    if err := cmd.Run(); err != nil {
        return "", err
    }
    
    // 3. 优化 PDF (可选：压缩、添加书签)
    if err := g.optimizePDF(pdfPath); err != nil {
        return "", err
    }
    
    return pdfPath, nil
}

func (g *PDFGenerator) renderHTML(report ReportData) (string, error) {
    // 使用 Go template 渲染 HTML
    tmpl, err := template.ParseFiles(filepath.Join(g.templateDir, "weekly-report.html"))
    if err != nil {
        return "", err
    }
    
    htmlPath := filepath.Join(g.outputDir, report.WeekID+".html")
    file, err := os.Create(htmlPath)
    if err != nil {
        return "", err
    }
    defer file.Close()
    
    if err := tmpl.Execute(file, report); err != nil {
        return "", err
    }
    
    return htmlPath, nil
}
```

### 3.2 Excel 生成代码

```go
package report

import (
    "github.com/xuri/excelize/v2"
)

type ExcelGenerator struct {
    templatePath string
}

func (g *ExcelGenerator) GenerateWeeklyReport(report ReportData) (string, error) {
    // 创建 Excel 文件
    f := excelize.NewFile()
    
    // Sheet 1: 核心指标
    index := f.NewSheet("核心指标")
    g.writeMetricsSheet(f, report.Metrics)
    g.addMetricsChart(f, report.MetricsTrend)
    
    // Sheet 2: 详细数据
    detail := f.NewSheet("详细数据")
    g.writeDetailSheet(f, report.DailyData)
    
    // Sheet 3: 团队贡献
    team := f.NewSheet("团队贡献")
    g.writeTeamSheet(f, report.TeamContributions)
    
    // 设置默认 Sheet
    f.SetActiveSheet(index)
    
    // 保存文件
    outputPath := filepath.Join(g.outputDir, report.WeekID+".xlsx")
    if err := f.SaveAs(outputPath); err != nil {
        return "", err
    }
    
    return outputPath, nil
}

func (g *ExcelGenerator) writeMetricsSheet(f *excelize.File, metrics MetricsData) {
    // 设置表头
    headers := []string{"指标名称", "本周值", "上周值", "变化量", "变化率", "目标值"}
    for i, header := range headers {
        cell, _ := excelize.CoordinatesToCellName(i+1, 1)
        f.SetCellValue("核心指标", cell, header)
        
        // 设置样式
        style, _ := f.NewStyle(&excelize.Style{
            Font: &excelize.Font{Bold: true},
            Fill: excelize.Fill{Type: "pattern", Color: []int{0xE0E0E0}},
        })
        f.SetCellStyle("核心指标", cell, cell, style)
    }
    
    // 填充数据
    for i, metric := range metrics {
        row := i + 2
        f.SetCellValue("核心指标", fmt.Sprintf("A%d", row), metric.Name)
        f.SetCellValue("核心指标", fmt.Sprintf("B%d", row), metric.CurrentValue)
        f.SetCellValue("核心指标", fmt.Sprintf("C%d", row), metric.PreviousValue)
        f.SetCellValue("核心指标", fmt.Sprintf("D%d", row), metric.Change)
        f.SetCellValue("核心指标", fmt.Sprintf("E%d", row), metric.ChangeRate)
        f.SetCellValue("核心指标", fmt.Sprintf("F%d", row), metric.Target)
    }
}

func (g *ExcelGenerator) addMetricsChart(f *excelize.File, trendData TrendData) {
    // 添加折线图
    chart := &excelize.Chart{
        Type: excelize.Line,
        Series: []excelize.ChartSeries{
            {
                Name: "部署频率",
                Categories: "详细数据!A2:A9",
                Values: "详细数据!B2:B9",
            },
        },
        Title: "DORA 指标趋势",
    }
    
    f.AddChart("核心指标", "H2", chart)
}
```

### 4.2 邮件发送

```go
package notification

import (
    "gopkg.in/gomail.v2"
)

type EmailSender struct {
    smtpHost string
    smtpPort int
    username string
    password string
}

func (s *EmailSender) SendWeeklyReport(recipients []string, report ReportData) error {
    // 生成 PDF 和 Excel 附件
    pdfPath, _ := pdfGenerator.GenerateWeeklyReport(report)
    xlsxPath, _ := excelGenerator.GenerateWeeklyReport(report)
    
    // 创建邮件
    m := gomail.NewMessage()
    m.SetHeader("From", s.username)
    m.SetHeader("To", recipients...)
    m.SetHeader("Subject", fmt.Sprintf("📊 研发效能周报 - %s (%s)", report.TeamName, report.WeekID))
    
    // HTML 正文
    htmlBody := s.generateEmailHTML(report)
    m.SetBody("text/html", htmlBody)
    
    // 添加附件
    m.Attach(pdfPath)
    m.Attach(xlsxPath)
    
    // 发送邮件
    d := gomail.NewDialer(s.smtpHost, s.smtpPort, s.username, s.password)
    if err := d.DialAndSend(m); err != nil {
        return err
    }
    
    return nil
}

func (s *EmailSender) generateEmailHTML(report ReportData) string {
    return fmt.Sprintf(`
<html>
<body>
    <h2>📊 %s · 研发效能周报</h2>
    <p><strong>周期</strong>: %s ~ %s</p>
    
    <h3>📈 核心指标</h3>
    <ul>
        <li>部署次数：<strong>%d 次</strong> (↑%d%% vs 上周)</li>
        <li>变更前置时间：<strong>%.1f 分钟</strong> (↓%.1f%% vs 上周)</li>
        <li>变更失败率：<strong>%.1f%%</strong> (↓%.1f%% vs 上周)</li>
        <li>MTTR: <strong>%.1f 分钟</strong> (↓%.1f%% vs 上周)</li>
    </ul>
    
    <p>详细报告请查看附件。</p>
    
    <p>——<br/>Orion 自动周报系统</p>
</body>
</html>
`, report.TeamName, report.StartDate, report.EndDate,
       report.Metrics.DeploymentCount, report.Metrics.DeploymentChange,
       report.Metrics.LeadTime, report.Metrics.LeadTimeChange,
       report.Metrics.FailureRate, report.Metrics.FailureRateChange,
       report.Metrics.MTTR, report.Metrics.MTTRChange)
```

## 6. API 设计

```yaml
# 周报导出 API
api:
  base_path: /api/v1/weekly-reports

  endpoints:
    # 导出
    - POST   /{id}/export                   # 导出周报
      params:
        - format: pdf|xlsx|html|md|png
        - include_charts: true|false
        - include_raw_data: true|false
    
    # 下载
    - GET    /{id}/download.{format}        # 下载指定格式
    
    # 分享
    - POST   /{id}/share                    # 生成分享链接
    - GET    /share/{token}                 # 访问分享页面
    - DELETE /share/{token}                 # 撤销分享
    
    # 发送
    - POST   /{id}/send                     # 发送邮件/钉钉
      params:
        - recipients: array
        - channels: array
        - message: string
```
