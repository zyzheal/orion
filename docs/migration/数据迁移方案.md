## 一、背景与目标

### 1.1 迁移背景

企业现有多个 DevOps 系统独立运行，存在数据孤岛问题：

| 现有系统 | 功能 | 数据库 | 数据量 |
|---------|------|--------|--------|
| Jenkins | CI/CD 流水线 | Jenkins Home (XML) | ~50GB |
| GitLab CE | 代码 + MR + CI | PostgreSQL 12 | ~200GB |
| 自研审批系统 | 发布审批 | MySQL 5.7 | ~10GB |
| 自研部署系统 | 生产部署 | MySQL 5.7 | ~5GB |
| 云效 (阿里) | 部分项目 CI | 阿里云 RDS | - |
| 自研监控系统 | 告警 + 看板 | InfluxDB + MySQL | ~50GB |

### 1.2 迁移目标

将现有系统数据迁移到 Orion 平台，实现：

1. **数据统一**：所有 DevOps 数据集中在 Orion 平台
2. **历史可追溯**：保留至少 2 年的历史数据用于分析
3. **业务连续性**：迁移过程不影响正常研发流程
4. **数据一致性**：确保迁移后数据完整、准确

### 1.3 迁移范围

| 数据类型 | 来源 | 目标 | 迁移量预估 | 优先级 |
|---------|------|------|-----------|--------|
| 用户数据 | GitLab + LDAP | Orion users | ~500 用户 | P0 |
| 项目数据 | GitLab | Orion product_lines | ~200 项目 | P0 |
| 流水线历史 | Jenkins + GitLab CI | Orion pipelines | ~10 万条记录 | P0 |
| 审批记录 | 自研审批系统 | Orion approvals | ~5 万条记录 | P0 |
| 部署记录 | 自研部署系统 | Orion deployments | ~2 万条记录 | P0 |
| 构建产物 | Nexus + Harbor | Orion artifacts (元数据) | ~1 万条元数据 | P1 |
| 监控告警 | 自研监控系统 | Orion notifications | ~100 万条 (聚合) | P1 |
| 效率指标 | 多系统 | Orion efficiency_metrics | ~50 万条 | P1 |

## 三、数据映射设计

### 3.1 用户数据映射

**来源**：GitLab users + LDAP

```sql
-- GitLab users 表结构
-- id, username, email, name, state, created_at, last_sign_in_at

-- Orion users 表结构
CREATE TABLE mapping_temp (
    source_system VARCHAR(32),    -- 来源系统：gitlab, ldap, legacy
    source_id     VARCHAR(64),    -- 原系统用户 ID
    target_id     CHAR(36),       -- Orion 用户 UUID
    mapping_status ENUM('pending', 'migrated', 'conflict', 'skipped'),
    conflict_reason TEXT,
    migrated_at   DATETIME
);

-- 映射规则
-- 1. 优先使用邮箱作为唯一标识
-- 2. 邮箱冲突时，使用 username + 系统后缀
-- 3. LDAP 用户优先，GitLab 用户补充
```

**迁移脚本**：

```python
#!/usr/bin/env python3
"""
用户数据迁移脚本
来源：GitLab API + LDAP
目标：Orion users 表
"""

import requests
import ldap
import uuid
from datetime import datetime
from typing import Dict, List, Optional

class UserMigrator:
    def __init__(self, config: dict):
        self.gitlab_url = config['gitlab']['url']
        self.gitlab_token = config['gitlab']['token']
        self.ldap_config = config['ldap']
        self.orion_db = config['orion_db']
        
    def fetch_gitlab_users(self) -> List[Dict]:
        """从 GitLab 获取用户列表"""
        users = []
        page = 1
        while True:
            resp = requests.get(
                f"{self.gitlab_url}/api/v4/users",
                params={'per_page': 100, 'page': page},
                headers={'PRIVATE-TOKEN': self.gitlab_token}
            )
            if resp.status_code != 200 or not resp.json():
                break
            users.extend(resp.json())
            page += 1
        return users
    
    def fetch_ldap_users(self) -> List[Dict]:
        """从 LDAP 获取用户列表"""
        conn = ldap.initialize(self.ldap_config['uri'])
        conn.simple_bind_s(self.ldap_config['bind_dn'], 
                          self.ldap_config['password'])
        
        search_filter = "(objectClass=person)"
        attrs = ['uid', 'mail', 'cn', 'sn']
        
        results = conn.search_s(
            self.ldap_config['base_dn'],
            ldap.SCOPE_SUBTREE,
            search_filter,
            attrs
        )
        
        users = []
        for dn, entry in results:
            if entry:
                users.append({
                    'uid': entry.get('uid', [b''])[0].decode(),
                    'email': entry.get('mail', [b''])[0].decode(),
                    'name': entry.get('cn', [b''])[0].decode(),
                })
        return users
    
    def migrate(self, dry_run: bool = True) -> Dict:
        """执行迁移"""
        stats = {'migrated': 0, 'conflicts': 0, 'skipped': 0}
        
        # 获取来源数据
        gitlab_users = self.fetch_gitlab_users()
        ldap_users = self.fetch_ldap_users()
        
        # 构建邮箱索引 (LDAP 优先)
        email_index = {}
        for user in ldap_users:
            email = user['email']
            email_index[email] = {**user, 'source': 'ldap'}
        
        # GitLab 用户补充
        for user in gitlab_users:
            email = user['email']
            if email not in email_index:
                email_index[email] = {**user, 'source': 'gitlab'}
        
        # 迁移到 Orion
        for email, user in email_index.items():
            if dry_run:
                print(f"[DRY RUN] Would migrate: {email}")
                stats['migrated'] += 1
                continue
            
            try:
                # 检查是否已存在
                cursor = self.orion_db.cursor()
                cursor.execute(
                    "SELECT id FROM users WHERE email = %s",
                    (email,)
                )
                if cursor.fetchone():
                    stats['skipped'] += 1
                    continue
                
                # 插入新用户
                user_id = str(uuid.uuid4())
                cursor.execute("""
                    INSERT INTO users (id, user_id, email, name, status, created_at)
                    VALUES (%s, %s, %s, %s, 'active', NOW())
                """, (user_id, user['uid'], email, user['name']))
                
                self.orion_db.commit()
                stats['migrated'] += 1
                
            except Exception as e:
                stats['conflicts'] += 1
                print(f"Error migrating {email}: {e}")
        
        return stats

if __name__ == '__main__':
    config = {
        'gitlab': {'url': 'https://gitlab.internal', 'token': 'xxx'},
        'ldap': {'uri': 'ldap://ldap.internal', 'bind_dn': 'cn=admin', 'password': 'xxx', 'base_dn': 'dc=company'},
        'orion_db': None  # MySQL connection
    }
    
    migrator = UserMigrator(config)
    stats = migrator.migrate(dry_run=True)
    print(f"Migration stats: {stats}")
```

**迁移脚本**：

```python
#!/usr/bin/env python3
"""
流水线历史数据迁移脚本
来源：Jenkins + GitLab CI
目标：Orion pipelines / pipeline_stages / pipeline_steps
"""

import xml.etree.ElementTree as ET
from datetime import datetime

class PipelineMigrator:
    def __init__(self, config: dict):
        self.jenkins_url = config['jenkins']['url']
        self.gitlab_url = config['gitlab']['url']
        self.orion_db = config['orion_db']
        
    def parse_jenkins_build(self, job_name: str, build_xml: str) -> Dict:
        """解析 Jenkins Build XML"""
        root = ET.fromstring(build_xml)
        
        return {
            'job_name': job_name,
            'build_number': root.find('number').text,
            'result': root.find('result').text if root.find('result') is not None else 'RUNNING',
            'duration': root.find('duration').text,
            'timestamp': root.find('timestamp').text,
            'built_on': root.find('builtOn').text if root.find('builtOn') is not None else 'master'
        }
    
    def migrate_jenkins_history(self, dry_run: bool = True) -> Dict:
        """迁移 Jenkins 构建历史"""
        stats = {'jobs': 0, 'builds': 0, 'errors': 0}
        
        # 获取所有 Job
        resp = requests.get(f"{self.jenkins_url}/api/json", 
                           params={'tree': 'jobs[name,url]'})
        jobs = resp.json().get('jobs', [])
        
        for job in jobs:
            job_name = job['name']
            stats['jobs'] += 1
            
            # 获取构建历史 (限制最近 2 年)
            build_range = datetime.now().timestamp() * 1000 - 365 * 2 * 24 * 60 * 60 * 1000
            
            build_resp = requests.get(
                f"{job['url']}/api/json",
                params={'tree': f'builds[number,result,duration,timestamp,id]{{0,500}}'}
            )
            builds = build_resp.json().get('builds', [])
            
            for build in builds:
                if build['timestamp'] < build_range:
                    break  # 超过 2 年的数据跳过
                
                if dry_run:
                    stats['builds'] += 1
                    continue
                
                try:
                    # 转换状态
                    status_map = {
                        'SUCCESS': 'success',
                        'FAILURE': 'failed',
                        'UNSTABLE': 'success',  # 不稳定视为成功
                        'ABORTED': 'cancelled',
                        'RUNNING': 'running'
                    }
                    
                    cursor = self.orion_db.cursor()
                    cursor.execute("""
                        INSERT INTO pipelines (
                            id, name, status, duration_ms, 
                            created_at, completed_at, source_system
                        ) VALUES (UUID(), %s, %s, %s, FROM_UNIXTIME(%s), 
                                 FROM_UNIXTIME(%s), 'jenkins')
                    """, (
                        f"{job_name} #{build['number']}",
                        status_map.get(build['result'], 'unknown'),
                        build['duration'],
                        build['timestamp'] / 1000,
                        (build['timestamp'] + build['duration']) / 1000
                    ))
                    
                    self.orion_db.commit()
                    stats['builds'] += 1
                    
                except Exception as e:
                    stats['errors'] += 1
                    print(f"Error migrating {job_name}#{build['number']}: {e}")
        
        return stats
```

```sql
-- 原审批系统表结构
CREATE TABLE legacy_approvals (
    id              BIGINT PRIMARY KEY,
    approval_type   VARCHAR(32),      -- 'deploy', 'change', 'emergency'
    title           VARCHAR(255),
    requester       VARCHAR(64),
    approvers       JSON,             -- ['user1', 'user2']
    status          VARCHAR(16),      -- 'pending', 'approved', 'rejected'
    comments        TEXT,
    created_at      DATETIME,
    completed_at    DATETIME
);

-- Orion approvals 表映射
INSERT INTO orion_public.approvals (
    id, approval_type, title, description,
    product_line_id, deployment_id,
    status, priority,
    requested_by, created_at,
    completed_at, completed_by
)
SELECT 
    UUID(),
    a.approval_type,
    a.title,
    a.comments,
    (SELECT id FROM product_lines LIMIT 1),  -- 需要关联
    (SELECT id FROM deployments WHERE legacy_id = a.id),
    CASE a.status
        WHEN 'pending' THEN 'pending'
        WHEN 'approved' THEN 'approved'
        WHEN 'rejected' THEN 'rejected'
    END,
    CASE a.approval_type
        WHEN 'emergency' THEN 'critical'
        ELSE 'normal'
    END,
    (SELECT id FROM users WHERE user_id = a.requester),
    a.created_at,
    a.completed_at,
    (SELECT id FROM users WHERE user_id = JSON_UNQUOTE(JSON_EXTRACT(a.approvers, '$[0]')))
FROM legacy_approvals a
WHERE a.created_at >= DATE_SUB(NOW(), INTERVAL 2 YEAR);
```

### 4.3 数据校验脚本

```python
#!/usr/bin/env python3
"""
数据迁移校验脚本
验证源系统与 Orion 数据一致性
"""

import mysql.connector
from typing import Dict, Tuple

class DataValidator:
    def __init__(self, source_config: dict, target_config: dict):
        self.source_db = mysql.connector.connect(**source_config)
        self.target_db = mysql.connector.connect(**target_config)
    
    def validate_user_count(self) -> Tuple[int, int, bool]:
        """校验用户数量"""
        source_cursor = self.source_db.cursor()
        target_cursor = self.target_db.cursor()
        
        source_cursor.execute("SELECT COUNT(*) FROM users WHERE state = 'active'")
        source_count = source_cursor.fetchone()[0]
        
        target_cursor.execute("SELECT COUNT(*) FROM users WHERE status = 'active'")
        target_count = target_cursor.fetchone()[0]
        
        passed = abs(source_count - target_count) < 5  # 允许 5% 误差
        return source_count, target_count, passed
    
    def validate_pipeline_count(self, days: int = 365) -> Tuple[int, int, bool]:
        """校验流水线数量 (最近 N 天)"""
        source_cursor = self.source_db.cursor()
        target_cursor = self.target_db.cursor()
        
        # Jenkins + GitLab CI
        source_cursor.execute("""
            SELECT COUNT(*) FROM (
                SELECT timestamp FROM jenkins_builds 
                WHERE timestamp >= UNIX_TIMESTAMP(DATE_SUB(NOW(), INTERVAL %s DAY))
                UNION ALL
                SELECT created_at FROM gitlab_pipelines
                WHERE created_at >= DATE_SUB(NOW(), INTERVAL %s DAY)
            ) t
        """, (days, days))
        source_count = source_cursor.fetchone()[0]
        
        target_cursor.execute("""
            SELECT COUNT(*) FROM pipelines
            WHERE created_at >= DATE_SUB(NOW(), INTERVAL %s DAY)
        """, (days,))
        target_count = target_cursor.fetchone()[0]
        
        passed = abs(source_count - target_count) / max(source_count, 1) < 0.05
        return source_count, target_count, passed
    
    def validate_approval_count(self) -> Tuple[int, int, bool]:
        """校验审批记录数量"""
        source_cursor = self.source_db.cursor()
        target_cursor = self.target_db.cursor()
        
        source_cursor.execute("""
            SELECT COUNT(*) FROM legacy_approvals
            WHERE created_at >= DATE_SUB(NOW(), INTERVAL 730 DAY)
        """)
        source_count = source_cursor.fetchone()[0]
        
        target_cursor.execute("""
            SELECT COUNT(*) FROM approvals
            WHERE created_at >= DATE_SUB(NOW(), INTERVAL 730 DAY)
        """)
        target_count = target_cursor.fetchone()[0]
        
        passed = abs(source_count - target_count) < 100
        return source_count, target_count, passed
    
    def sample_validation(self, table: str, sample_size: int = 10) -> Dict:
        """抽样验证"""
        target_cursor = self.target_db.cursor(dictionary=True)
        target_cursor.execute(f"SELECT * FROM {table} ORDER BY RAND() LIMIT %s", (sample_size,))
        samples = target_cursor.fetchall()
        
        issues = []
        for row in samples:
            # 检查关键字段是否为 NULL
            for key, value in row.items():
                if key.endswith('_at') and value is None:
                    issues.append(f"{table}.{key} is NULL for id={row.get('id')}")
                if key in ['email', 'name'] and not value:
                    issues.append(f"{table}.{key} is empty for id={row.get('id')}")
        
        return {'total': sample_size, 'issues': len(issues), 'details': issues}
    
    def run_all_validations(self) -> Dict:
        """运行所有校验"""
        results = {}
        
        results['users'] = self.validate_user_count()
        results['pipelines'] = self.validate_pipeline_count()
        results['approvals'] = self.validate_approval_count()
        results['user_sample'] = self.sample_validation('users')
        results['pipeline_sample'] = self.sample_validation('pipelines')
        results['approval_sample'] = self.sample_validation('approvals')
        
        # 总体评估
        all_passed = all([
            results['users'][2],
            results['pipelines'][2],
            results['approvals'][2],
            results['user_sample']['issues'] == 0,
            results['pipeline_sample']['issues'] == 0,
            results['approval_sample']['issues'] == 0
        ])
        
        results['overall_passed'] = all_passed
        return results

if __name__ == '__main__':
```
