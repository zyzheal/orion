```go
// orion-go-common/pkg/auth/jwt.go (新增)
type JWTConfig struct {
    // RS256 for access tokens
    PrivateKeyPath string
    PublicKeyPath  string
    // HS256 for refresh tokens (backward compat)
    HS256Secret    string
    AccessTTL      time.Duration
    RefreshTTL     time.Duration
}

func (j *JWTConfig) SignAccessToken(claims *Claims) (string, error) {
    // RS256 signing
    token := jwt.NewWithClaims(jwt.SigningMethodRS256, claims)
    return token.SignedString(j.privateKey)
}

// orion-go-common/pkg/auth/middleware.go (修改)
// 验证时支持双算法
token, err := jwt.Parse(tokenString, func(token *jwt.Token) (interface{}, error) {
    switch token.Method.(type) {
    case *jwt.SigningMethodRSA:
        return cfg.PublicKey, nil
    case *jwt.SigningMethodHMAC:
        return []byte(cfg.HS256Secret), nil
    default:
        return nil, jwt.ErrSignatureInvalid
    }
})
```

**1.4 HttpOnly Cookie 核心变更**：

```go
// orion-auth-svc/internal/handler/handler.go (修改 Login)
func (h *Handler) Login(c *gin.Context) {
    // ... 验证逻辑 ...
    tokens, err := h.services.JWT.GenerateTokens(user.ID, user.TenantID, user.Role)

    // 设置 HttpOnly Cookie 而非返回 JSON body
    c.SetSameSite(http.SameSiteStrictMode)
    c.SetCookie("access_token", tokens.AccessToken, int(h.cfg.JWTExpiration.Seconds()),
        "/", "", cfg.SecureCookie, true) // HttpOnly=true
    c.SetCookie("refresh_token", tokens.RefreshToken, int(h.cfg.JWTRefreshExpiration.Seconds()),
        "/", "", cfg.SecureCookie, true)

    c.JSON(http.StatusOK, gin.H{"expires_in": tokens.ExpiresIn, "token_type": "Bearer"})
}
```

**1.5 并发刷新保护**：

```go
// orion-auth-svc/internal/handler/handler.go (修改 RefreshToken)
const refreshLua = `
local current = redis.call('GET', KEYS[1])
if current == ARGV[1] then
    redis.call('SET', KEYS[1], ARGV[2], 'EX', ARGV[3])
    return 'OK'
else
    redis.call('DEL', KEYS[1])
    return 'REVOKED'
end
`

func (h *Handler) RefreshToken(c *gin.Context) {
    // ... 提取 refresh_token from Cookie ...
    result, err := h.rdb.Eval(c.Request.Context(), refreshLua,
        []string{"refresh:" + claims.JTI}, oldToken, newToken, ttl).Result()
    if result == "REVOKED" {
        // 并发刷新检测，全部吊销
        h.rdb.Del(c.Request.Context(), "refresh:"+claims.JTI)
        c.AbortWithStatusJSON(401, gin.H{"error": "concurrent refresh detected, all tokens revoked"})
        return
    }
}
```

```go
// orion-go-common/pkg/database/repository.go (新增)
func (r *BaseRepository) FindOne(ctx context.Context, table, tenantID, id string, dest interface{}) error {
    if err := validateIdentifier(table); err != nil {
        return err
    }
    query := fmt.Sprintf("SELECT * FROM %s WHERE id = $1 AND tenant_id = $2 AND status != 'deleted'", table)
    return r.db.GetContext(ctx, dest, query, id, tenantID)
}

func (r *BaseRepository) FindList(ctx context.Context, table, tenantID, where string, dest interface{}, args ...interface{}) error {
    if err := validateIdentifier(table); err != nil {
        return err
    }
    query := fmt.Sprintf("SELECT * FROM %s WHERE tenant_id = $1 AND status != 'deleted' AND %s", table, where)
    allArgs := append([]interface{}{tenantID}, args...)
    return r.db.SelectContext(ctx, dest, query, allArgs...)
}
```

```go
// orion-go-common/pkg/audit/chain.go (新增)
type AuditEntry struct {
    ID        string
    PrevHash  string
    Data      []byte
    Hash      string
    Timestamp time.Time
}

func ComputeHash(entry *AuditEntry) string {
    h := sha256.New()
    h.Write([]byte(entry.PrevHash))
    h.Write(entry.Data)
    h.Write([]byte(entry.Timestamp.Format(time.RFC3339Nano)))
    return hex.EncodeToString(h.Sum(nil))
}
```
