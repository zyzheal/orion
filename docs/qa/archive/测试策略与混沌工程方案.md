## 一、测试策略总览

### 1.1 测试金字塔

```
                    ┌─────────────┐
                   /    E2E 测试    \
                  /    (10%)        \
                 /───────────────────\
                /   集成测试          \
               /    (20%)            \
              /───────────────────────\
             /    单元测试             \
            /    (70%)                 \
           /───────────────────────────\
```

### 1.2 测试类型矩阵

| 测试类型 | 占比 | 执行时机 | 目标时间 | 负责人 |
|---------|------|---------|---------|--------|
| 单元测试 | 70% | CI 流水线 | < 5 分钟 | 开发 |
| 集成测试 | 15% | CI 流水线 | < 10 分钟 | 开发/测试 |
| E2E 测试 | 10% | 预发布环境 | < 30 分钟 | 测试 |
| 性能测试 | 3% | 独立环境 | < 2 小时 | 测试/SRE |
| 混沌测试 | 2% | 定期演练 | < 1 小时 | SRE/测试 |

### 1.3 测试覆盖率目标

| 模块类型 | 行覆盖率 | 分支覆盖率 | 关键路径覆盖率 |
|---------|---------|-----------|--------------|
| 核心服务 (API/权限) | > 90% | > 85% | 100% |

package api

import (
    "testing"
    "github.com/stretchr/testify/assert"

type MockPipelineService struct {
    mock.Mock
}

func (m *MockPipelineService) GetPipeline(id string) (*Pipeline, error) {
    args := m.Called(id)
    return args.Get(0).(*Pipeline), args.Error(1)
}

// 测试：成功获取流水线
func TestGetPipeline_Success(t *testing.T) {
    // Arrange
    mockService := new(MockPipelineService)
    expectedPipeline := &Pipeline{
        ID:   "pl-123",
        Name: "test-pipeline",
    }
    mockService.On("GetPipeline", "pl-123").Return(expectedPipeline, nil)
    
    handler := NewPipelineHandler(mockService)
    
    // Act
    result, err := handler.GetPipeline("pl-123")
    
    // Assert
    assert.NoError(t, err)
    assert.Equal(t, expectedPipeline, result)
    mockService.AssertExpectations(t)
}

// 测试：流水线不存在
func TestGetPipeline_NotFound(t *testing.T) {
    // Arrange
    mockService := new(MockPipelineService)
    mockService.On("GetPipeline", "non-existent").Return(nil, ErrPipelineNotFound)
    
    handler := NewPipelineHandler(mockService)
    
    // Act
    result, err := handler.GetPipeline("non-existent")
    
    // Assert
    assert.Error(t, err)
    assert.Equal(t, ErrPipelineNotFound, err)
    assert.Nil(t, result)
    mockService.AssertExpectations(t)
}

    { duration: '5m', target: 0 },     // 逐步减少
  ],

    http_req_duration: ['p(95)<500'],  // P95 < 500ms
    http_req_failed: ['rate<0.01'],    // 错误率 < 1%
    errors: ['rate<0.05'],             // 自定义错误率 < 5%
    api_latency: ['avg<200'],          // 平均延迟 < 200ms
  },
}

export default function () {
  // 场景 1: 流水线列表
  const res1 = http.get('http://orion-api/api/v1/pipelines')
  check(res1, {
    'pipeline list: status is 200': (r) => r.status === 200,
    'pipeline list: has data': (r) => JSON.parse(r.body).data.items.length >= 0,
  })
  errorRate.add(res1.status !== 200)
  apiLatency.add(res1.timings.duration)
  
  sleep(1)
  
  // 场景 2: 触发流水线
  const res2 = http.postJson(
    'http://orion-api/api/v1/pipelines/pl-001/run',

    { headers: { 'Content-Type': 'application/json' } }
  )
  check(res2, {
    'trigger pipeline: status is 201': (r) => r.status === 201,
  })
  errorRate.add(res2.status !== 201)
  apiLatency.add(res2.timings.duration)
  
  sleep(2)
}
